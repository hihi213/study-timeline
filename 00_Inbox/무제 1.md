-- 1-1) 트리거 생성(템플릿)
CREATE OR REPLACE TRIGGER <TRG_NAME>
    AFTER UPDATE OR DELETE ON <TABLE_NAME>
    FOR EACH ROW
DECLARE
    v_modeType VARCHAR2(10);
BEGIN
    -- 동작 구분(시스템 조건)
    IF UPDATING THEN
        v_modeType := '수정';
    ELSIF DELETING THEN
        v_modeType := '삭제';
    END IF;

    -- 변경 전(:OLD) + 실행자(USER) 기록
    INSERT INTO backup_userTBL (userID, userName, modType, modDate, modUser)
    VALUES (:OLD.userID, :OLD.userName, v_modeType, SYSDATE, USER);
END;

ON customer(CASE WHEN phone IS NOT NULL THEN phone END);

-- 4-1) 시퀀스 생성(로그 ID 자동 증가)
CREATE SEQUENCE audit_log_seq
START WITH 1
INCREMENT BY 1
NOCACHE
NOCYCLE;

-- 4-2) 감사 로그 테이블 생성
CREATE TABLE audit_log (
    log_id        NUMBER PRIMARY KEY,
    username      VARCHAR2(50),
    access_time   DATE,
    action_type   VARCHAR2(10),   -- INSERT/UPDATE/DELETE
    accessed_col  VARCHAR2(50),   -- SSN/SALARY 등
    accessed_data VARCHAR2(200)   -- 이전값 -> 새값
);

-- 4-3) 로그 ID 자동 입력 트리거
CREATE OR REPLACE TRIGGER trg_audit_log_id
BEFORE INSERT ON audit_log
FOR EACH ROW
BEGIN
    :NEW.log_id := audit_log_seq.NEXTVAL;
END;
/

-- 4-4) SSN 수정 감시 트리거(UPDATE OF ssn)
CREATE OR REPLACE TRIGGER trg_audit_ssn_update
AFTER UPDATE OF ssn ON employee2
FOR EACH ROW
BEGIN
    INSERT INTO audit_log (username, access_time, action_type, accessed_col, accessed_data)
    VALUES (USER, SYSDATE, 'UPDATE', 'SSN', :OLD.ssn || ' -> ' || :NEW.ssn);
END;
/

-- 4-5) SSN 삭제 감시 트리거(BEFORE DELETE)
CREATE OR REPLACE TRIGGER trg_audit_ssn_delete
BEFORE DELETE ON employee2
FOR EACH ROW
BEGIN
    INSERT INTO audit_log (username, access_time, action_type, accessed_col, accessed_data)
    VALUES (USER, SYSDATE, 'DELETE', 'SSN', :OLD.ssn);
END;
/

WHEN (NEW.salary > 10000)

END;
/

CREATE OR REPLACE TRIGGER trg_audit_salary_insert
AFTER INSERT ON employee2
FOR EACH ROW
WHEN (NEW.salary > 10000)
BEGIN
    INSERT INTO audit_log (username, access_time, action_type, accessed_col, accessed_data)
    VALUES (USER, SYSDATE, 'INSERT', 'SALARY', :NEW.salary);
END;

   9) 트리거(Trigger) 템플릿 보충: :OLD/:NEW, 시점/이벤트 조합, INSTEAD OF
   ========================================================= */

-- 9-1) 기본 템플릿(:OLD/:NEW 사용하려면 FOR EACH ROW 필수)
CREATE OR REPLACE TRIGGER <TRG_NAME>
    BEFORE INSERT OR UPDATE OR DELETE ON <TABLE_NAME>
    FOR EACH ROW
BEGIN
    IF INSERTING THEN
        -- :NEW.<col> 사용 가능
        NULL;
    ELSIF UPDATING THEN
        -- :OLD.<col>, :NEW.<col> 비교 가능
        NULL;
    ELSIF DELETING THEN
        -- :OLD.<col> 사용 가능 (:NEW는 보통 의미 없음)
        NULL;
    END IF;
END;
/

-- 9-2) VIEW에 대한 DML을 대신 처리하는 INSTEAD OF 트리거(개념/문법 포인트)
-- (뷰에 INSERT/UPDATE/DELETE가 들어오면 실제 테이블에 반영)
CREATE OR REPLACE TRIGGER trg_vw_instead
INSTEAD OF INSERT OR UPDATE OR DELETE ON <VIEW_NAME>
FOR EACH ROW
BEGIN
    IF INSERTING THEN
        INSERT INTO <BASE_TABLE>(col1, col2) VALUES (:NEW.col1, :NEW.col2);
    ELSIF UPDATING THEN
        UPDATE <BASE_TABLE>
           SET col2 = :NEW.col2
         WHERE col1 = :OLD.col1;
    ELSIF DELETING THEN
        DELETE FROM <BASE_TABLE> WHERE col1 = :OLD.col1;
    END IF;
END;
/

-- 전제: salesman_1이 hr.DEPARTMENTS를 조회하려면 "hr."를 붙여야 함
-- 해결: (salesman_1 스키마에) 동의어 생성

CONN salesman_1/1234;

-- 개인 동의어(Private Synonym)
CREATE SYNONYM departments FOR hr.departments;

-- 이제 소유자 명시 없이 가능
SELECT * FROM departments;

-- 동의어 삭제
DROP SYNONYM departments;

-- (참고) 공용 동의어(Public Synonym) - 보통 DBA가 관리
CONN / AS SYSDBA;
CREATE PUBLIC SYNONYM departments FOR hr.departments;
-- 공용 동의어 삭제
DROP PUBLIC SYNONYM departments;

/* =========================================================
   19) "테이블은 숨기고 뷰만 공개" 권한 부여 패턴(보안 뷰 활용)
   ========================================================= */

-- (HR) 민감 컬럼 제외한 뷰 생성
CONN hr/1234;

CREATE OR REPLACE VIEW vw_emp_public AS
SELECT emp_id, name, department, salary
FROM employees;

-- (HR) 사용자에게는 뷰만 SELECT 권한 부여
GRANT SELECT ON vw_emp_public TO salesman_1;

-- (권장) 실제 테이블 권한은 안 줌 / 또는 회수
REVOKE SELECT ON employees FROM salesman_1;

-- (salesman_1) 뷰 조회
CONN salesman_1/1234;
SELECT * FROM hr.vw_emp_public;

/* =========================================================
   20) 롤(Role) 추가 실습 보충: 활성화/비활성화, 기본 롤 설정
   ========================================================= */

-- 20-1) 롤 부여/회수
CONN / AS SYSDBA;

GRANT team1 TO user1;
REVOKE team1 FROM user1;

-- 20-2) (사용자 세션에서) 롤 활성화/비활성화
CONN user1/1234;

-- 현재 활성 롤 확인(환경에 따라 뷰/명령 사용)
-- SQL*Plus에서는 보통 아래로 확인(권장):
SELECT * FROM USER_ROLE_PRIVS;

-- 특정 롤만 활성화
SET ROLE team1;

-- 모든 롤 비활성화(권한 테스트용)
SET ROLE NONE;

-- 20-3) 기본 롤(Default Role) 지정(관리자/권한자)
CONN / AS SYSDBA;

-- user1 로그인 시 team1만 기본 활성
ALTER USER user1 DEFAULT ROLE team1;

-- user1 로그인 시 모든 부여된 롤 기본 활성
ALTER USER user1 DEFAULT ROLE ALL;

-- 기본 롤에서 특정 롤 제외
ALTER USER user1 DEFAULT ROLE ALL EXCEPT team1;

/* =========================================================
   21) 계정 잠금/해제 & 비밀번호 변경(프로파일 실습과 같이 자주 출제)
   ========================================================= */

CONN / AS SYSDBA;

-- 계정 잠금
ALTER USER salesman_1 ACCOUNT LOCK;

-- 계정 잠금 해제 + 비밀번호 변경
ALTER USER salesman_1 ACCOUNT UNLOCK IDENTIFIED BY 1234;

/* =========================================================
   22) (검증) "권한이 부여됐는지" 사용자 관점에서 빠르게 확인하는 뷰들
   ========================================================= */

-- (사용자) 내가 가진 시스템 권한
SELECT * FROM USER_SYS_PRIVS;

-- (사용자) 내가 가진 오브젝트 권한(내가 받은 권한)
SELECT * FROM USER_TAB_PRIVS_RECD;

-- (사용자) 내가 남에게 준 오브젝트 권한(내가 준 권한)
SELECT * FROM USER_TAB_PRIVS_MADE;

-- (사용자) 내가 가진 롤
SELECT * FROM USER_ROLE_PRIVS;
