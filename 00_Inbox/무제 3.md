```python
# ✅ 반드시 Pydantic 모델 사용
from pydantic import BaseModel, Field, field_validator
from datetime import datetime
from typing import Optional

class BidNotice(BaseModel):
    notice_id: str = Field(..., description="입찰공고 번호")
    title: str
    organization: str
    amount: int  # 정규화된 금액 (원 단위)
    date: datetime  # 파싱된 날짜
    region: Optional[str] = None
    
    @field_validator('amount', mode='before')
    @classmethod
    def normalize_amount(cls, v):
        """금액 문자열 -> int 변환 (예: '1,000,000원' -> 1000000)"""
        if isinstance(v, str):
            return int(v.replace(',', '').replace('원', '').strip())
        return v
    
    @field_validator('date', mode='before')
    @classmethod
    def parse_date(cls, v):
        """날짜 문자열 -> datetime 변환"""
        if isinstance(v, str):
            return datetime.strptime(v, '%Y-%m-%d')
        return v
    
    class Config:
        json_encoders = {
            datetime: lambda v: v.isoformat()
        }
```

```python
# src/utils/exceptions.py
class ScrapingError(Exception):
    """크롤링 관련 기본 예외"""
    pass

class NetworkError(ScrapingError):
    """네트워크 오류 (재시도 가능)"""
    pass

class ParsingError(ScrapingError):
    """파싱 실패 (해당 항목 스킵)"""
    pass

# src/scraper/driver.py
from tenacity import retry, stop_after_attempt, wait_exponential

class PlaywrightDriver:
    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=2, max=10),
        retry=retry_if_exception_type(NetworkError),
        before_sleep=lambda retry_state: logger.warning(
            "retrying", attempt=retry_state.attempt_number
        )
    )
    async def fetch_page(self, url: str) -> str:
        try:
            await self.page.goto(url, timeout=10000)
            return await self.page.content()
        except TimeoutError as e:
            raise NetworkError(f"Page load timeout: {url}") from e
```

**Graceful Degradation 전략**:
- 상세 페이지 파싱 실패 → 해당 공고만 스킵 후 계속 진행
- 네트워크 에러 → 3회 재시도 후 checkpoint 저장하고 종료
- 필수 필드 누락 → Pydantic ValidationError 로깅 후 스킵

```python
from pathlib import Path
import json

class CheckpointManager:
    def __init__(self, checkpoint_path: Path = Path("data/checkpoint.json")):
        self.path = checkpoint_path
        
    def save(self, last_notice_id: str, page: int) -> None:
        """마지막 처리 지점 저장"""
        data = {
            "last_notice_id": last_notice_id,
            "page": page,
            "timestamp": datetime.now().isoformat()
        }
        self.path.write_text(json.dumps(data, indent=2))
        logger.info("checkpoint_saved", **data)
    
    def load(self) -> Optional[dict]:
        """재시작 지점 로드"""
        if not self.path.exists():
```

### 5. 중복 방지 (Application + DB 레벨)

```python
# src/repository/csv_repository.py
class CSVRepository(BidRepository):
    def __init__(self, csv_path: Path):
        self.csv_path = csv_path
        self._load_existing_ids()
    
    def _load_existing_ids(self) -> None:
        """메모리에 기존 ID 세트 캐시"""
        if self.csv_path.exists():
            df = pd.read_csv(self.csv_path)
            self._existing_ids = set(df['notice_id'].values)
        else:
            self._existing_ids = set()
    
    def exists(self, notice_id: str) -> bool:
        """O(1) 중복 체크"""
        return notice_id in self._existing_ids
    
    def save(self, notice: BidNotice) -> None:
        """중복 체크 후 저장"""
        if self.exists(notice.notice_id):
            logger.debug("duplicate_skipped", notice_id=notice.notice_id)
            return
        
        # CSV 추가
        df = pd.DataFrame([notice.model_dump()])
        df.to_csv(self.csv_path, mode='a', header=not self.csv_path.exists(), index=False)
        self._existing_ids.add(notice.notice_id)
        logger.info("notice_saved", notice_id=notice.notice_id)
```

```python
# src/utils/config.py
from pydantic_settings import BaseSettings
import yaml

class Config(BaseSettings):
    scraping_base_url: str
    scraping_wait_timeout: int
    # ... Pydantic로 검증
    
    @classmethod
    def from_yaml(cls, path: str = "config.yaml"):
        with open(path) as f:
            data = yaml.safe_load(f)
        # Flatten nested dict
        flat = {}
        for section, values in data.items():
            for key, value in values.items():
                flat[f"{section}_{key}"] = value
        return cls(**flat)

config = Config.from_yaml()
```

### 7. CLI 인터페이스 (Click 사용)

```python
# cli.py
import click
from src.service.scraping_service import ScrapingService
from src.utils.config import config

@click.command()
@click.option('--keyword', '-k', required=True, help='검색 키워드')
@click.option('--pages', '-p', default=5, help='수집할 페이지 수')
@click.option('--mode', type=click.Choice(['once', 'resume', 'interval']), default='once')
@click.option('--interval', default=60, help='interval 모드의 실행 주기(분)')
def main(keyword: str, pages: int, mode: str, interval: int):
    """누리장터 입찰공고 크롤러"""
    service = ScrapingService(...)
    
    if mode == 'once':
        asyncio.run(service.scrape(keyword, pages))
    elif mode == 'resume':
        asyncio.run(service.scrape(keyword, pages, resume=True))
    elif mode == 'interval':
        # APScheduler로 주기 실행
        from apscheduler.schedulers.blocking import BlockingScheduler
        scheduler = BlockingScheduler()
        scheduler.add_job(
            lambda: asyncio.run(service.scrape(keyword, pages)),
            'interval',
            minutes=interval
        )
        scheduler.start()

if __name__ == '__main__':
    main()
```

## 💡 개발자 역할 (AI가 못하는 것)

AI가 코드를 생성해도, **반드시 개발자가 직접 해야 할 것**:

1. **실제 사이트 테스트**: 누리장터에서 돌려보고 에러 확인
2. **설계 의도 이해**: "왜 이렇게 짰지?" 이해 못 하면 면접 답변 불가
3. **Edge Case 발견**: "공고 번호 없으면?", "페이지 끝이면?" 시나리오 추가
4. **회고 작성**: 각 기능 완성 후 "뭘 배웠는지" 메모
5. **README 검증**: 다른 환경(깨끗한 가상환경)에서 실행 테스트
