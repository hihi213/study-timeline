-- PostgreSQL ≥ 13
BEGIN;

-- =========================================================
-- 0) 확장 (대소문자 비구분 문자열, 유사검색, EXCLUDE 인덱스용)
-- =========================================================
CREATE EXTENSION IF NOT EXISTS citext;
CREATE EXTENSION IF NOT EXISTS pg_trgm;
CREATE EXTENSION IF NOT EXISTS btree_gist;

-- =========================================================
-- 1) 계정·권한 (RBAC)
-- =========================================================
CREATE TABLE users (
  user_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  username       CITEXT NOT NULL UNIQUE,                 -- 로그인 ID(대소문자 구분 없음)
  email          CITEXT UNIQUE,
  password_hash  TEXT   NOT NULL,
  display_name   TEXT   NOT NULL,
  gender         TEXT   CHECK (gender IN ('M','F')),     -- 층별 접근 정책(2F=남, 3~5F=여) 검증에 사용
  is_active      BOOLEAN NOT NULL DEFAULT TRUE,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE roles (
  role_id  SMALLINT PRIMARY KEY,
  name     TEXT NOT NULL UNIQUE CHECK (name IN ('ADMIN_ROOT','ADMIN','INSPECTOR','RESIDENT'))
);
INSERT INTO roles(role_id,name) VALUES (0,'ADMIN_ROOT'),(1,'ADMIN'),(2,'INSPECTOR'),(3,'RESIDENT')
ON CONFLICT DO NOTHING;

CREATE TABLE user_roles (
  user_id BIGINT   NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  role_id SMALLINT NOT NULL REFERENCES roles(role_id),
  PRIMARY KEY (user_id, role_id)
);

-- =========================================================
-- 2) 건물·거주 (층/호/개인번호, 활성 1개 제약)
-- =========================================================
CREATE TABLE floors (
  floor_no SMALLINT PRIMARY KEY,                          -- 2,3,4,5
  gender   TEXT NOT NULL CHECK (gender IN ('M','F'))      -- 2F=M, 3~5F=F
);
INSERT INTO floors(floor_no,gender) VALUES (2,'M'),(3,'F'),(4,'F'),(5,'F')
ON CONFLICT DO NOTHING;

CREATE TABLE rooms (
  room_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  floor_no  SMALLINT NOT NULL REFERENCES floors(floor_no),
  room_no   SMALLINT NOT NULL CHECK (room_no BETWEEN 1 AND 24),
  capacity  SMALLINT NOT NULL DEFAULT 3,                  -- 1인실은 1
  is_single BOOLEAN NOT NULL DEFAULT FALSE,
  UNIQUE(floor_no, room_no)
);

CREATE TABLE residencies (
  residency_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id      BIGINT  NOT NULL REFERENCES users(user_id),
  room_id      BIGINT  NOT NULL REFERENCES rooms(room_id),
  personal_no  SMALLINT NOT NULL CHECK (personal_no BETWEEN 1 AND 3),
  start_date   DATE    NOT NULL,
  end_date     DATE,                                       -- 중도 퇴사 시 세팅
  CHECK (end_date IS NULL OR end_date >= start_date)
);
-- 동일 방-개인번호 활성 중복 금지
CREATE UNIQUE INDEX ux_room_person_active
  ON residencies(room_id, personal_no) WHERE end_date IS NULL;
-- 사용자 활성 거주 1개
CREATE UNIQUE INDEX ux_user_active_residency
  ON residencies(user_id) WHERE end_date IS NULL;

-- 층별장(동시 2명 운용)
CREATE TABLE floor_head_assignments (
  assign_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  floor_no    SMALLINT NOT NULL REFERENCES floors(floor_no),
  user_id     BIGINT   NOT NULL REFERENCES users(user_id),
  slot        SMALLINT NOT NULL CHECK (slot IN (1,2)),     -- 동시 검사/관리 구분 슬롯
  assigned_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  revoked_at  TIMESTAMPTZ
);
CREATE UNIQUE INDEX ux_floor_slot_active
  ON floor_head_assignments(floor_no, slot) WHERE revoked_at IS NULL;
CREATE UNIQUE INDEX ux_floor_head_user_active
  ON floor_head_assignments(floor_no, user_id) WHERE revoked_at IS NULL;

-- =========================================================
-- 3) 공통 자원 (Laundry/Compartment/다목적실 통합 키)
--    * 도서관은 별도 도메인으로 분리. 필요 시 소장본을 자원으로 연결 가능.
-- =========================================================
CREATE TABLE resources (
  resource_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  kind          TEXT NOT NULL CHECK (kind IN (
                   'FRIDGE_COMPARTMENT','LAUNDRY_WASHER','LAUNDRY_DRYER','LAUNDRY_DUVET_DRYER','MULTIROOM'
                 )),
  name          TEXT NOT NULL,                                    -- 표기명
  floor_no      SMALLINT REFERENCES floors(floor_no),             -- 세탁실/칸의 소속 층(공용은 NULL)
  access_scope  TEXT NOT NULL DEFAULT 'ALL'
                CHECK (access_scope IN ('MEN','WOMEN','ALL')),    -- 이용 주체 제한
  display_order SMALLINT,                                         -- UI 정렬
  attrs         JSONB  NOT NULL DEFAULT '{}'::jsonb,              -- 소소한 속성(기본시간 등)
  is_active     BOOLEAN NOT NULL DEFAULT TRUE,
  UNIQUE(kind, floor_no, name)
);

-- 현재 상태 스냅샷(읽기 빠름) + 이벤트 로그(감사 추적)
CREATE TABLE resource_status (
  resource_id BIGINT PRIMARY KEY REFERENCES resources(resource_id) ON DELETE CASCADE,
  status      TEXT NOT NULL CHECK (status IN (
                'AVAILABLE','IN_USE','DONE','UNAUTHORIZED_USE','ERROR_REPORTED','OUT_OF_SERVICE'
              )),
  since       TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE resource_status_events (
  event_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  resource_id   BIGINT NOT NULL REFERENCES resources(resource_id) ON DELETE CASCADE,
  from_status   TEXT,
  to_status     TEXT NOT NULL,
  actor_user_id BIGINT REFERENCES users(user_id),                 -- 시스템 변경 시 NULL
  payload       JSONB NOT NULL DEFAULT '{}'::jsonb,               -- {reason, note, ...}
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ix_rse_resource ON resource_status_events(resource_id, created_at DESC);

-- 자원 사용 세션(남은시간 정정/미등록 사용/강제 AVAILABLE 근거)
CREATE TABLE resource_use_sessions (
  session_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  resource_id     BIGINT NOT NULL REFERENCES resources(resource_id) ON DELETE CASCADE,
  user_id         BIGINT REFERENCES users(user_id),
  started_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  expected_end_at TIMESTAMPTZ,
  ended_at        TIMESTAMPTZ,
  state           TEXT NOT NULL CHECK (state IN ('IN_USE','DONE','CANCELLED')),
  meta            JSONB NOT NULL DEFAULT '{}'::jsonb               -- {adjustments:[{by,deltaMin,at}], ...}
);
CREATE INDEX ix_rus_resource ON resource_use_sessions(resource_id, started_at DESC);

-- 공통 예약(다목적실 등). BOOKED 간 겹침 금지.
CREATE TABLE reservations (
  reservation_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  resource_id    BIGINT NOT NULL REFERENCES resources(resource_id) ON DELETE CASCADE,
  user_id        BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  period         TSTZRANGE NOT NULL,                               -- [start, end)
  status         TEXT NOT NULL CHECK (status IN ('BOOKED','CANCELLED','NOSHOW','COMPLETED')),
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT chk_resv_period_valid CHECK (lower(period) < upper(period))
);
CREATE INDEX ix_reservations_gist ON reservations USING gist (resource_id, period);
ALTER TABLE reservations
  ADD CONSTRAINT ex_reservations_no_overlap
  EXCLUDE USING gist (resource_id WITH =, period WITH &&)
  WHERE (status = 'BOOKED');

-- =========================================================
-- 4) 냉장고 (층당 1대, 칸=냉장*3 + 냉동*1)
--    * 칸=자원(resources.kind='FRIDGE_COMPARTMENT')로 매핑
-- =========================================================
CREATE TABLE fridges (
  fridge_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  floor_no   SMALLINT NOT NULL UNIQUE REFERENCES floors(floor_no),
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE compartments (
  compartment_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  fridge_id      BIGINT NOT NULL REFERENCES fridges(fridge_id) ON DELETE CASCADE,
  position_order SMALLINT NOT NULL CHECK (position_order BETWEEN 1 AND 4),  -- 1~3=냉장, 4=냉동
  kind           TEXT NOT NULL CHECK (kind IN ('FRIDGE','FREEZER')),
  name           TEXT,                                                       -- 표기용(예: 냉장1, 냉동)
  resource_id    BIGINT                                                      -- 생성 후 resources와 1:1 연결
  -- 고정 제약은 시드 이후 아래에서 추가
);

-- --- 시드: 각 층별 냉장고 1대 + 칸 4개 생성
INSERT INTO fridges(floor_no) SELECT f.floor_no FROM floors f
ON CONFLICT DO NOTHING;

INSERT INTO compartments (fridge_id, position_order, kind, name)
SELECT fr.fridge_id, pos,
       CASE WHEN pos <= 3 THEN 'FRIDGE' ELSE 'FREEZER' END AS kind,
       CASE WHEN pos <= 3 THEN CONCAT('냉장',pos) ELSE '냉동' END AS name
FROM fridges fr
CROSS JOIN generate_series(1,4) AS pos
ON CONFLICT DO NOTHING;

-- --- 칸을 공통 자원으로 등록하고 1:1 매핑
WITH ins AS (
  INSERT INTO resources(kind, name, floor_no, display_order, access_scope)
  SELECT 'FRIDGE_COMPARTMENT',
         c.name,
         f.floor_no,
         c.position_order,
         'ALL'
  FROM compartments c JOIN fridges f ON f.fridge_id = c.fridge_id
  ON CONFLICT DO NOTHING
  RETURNING resource_id, floor_no, display_order
)
UPDATE compartments c
SET resource_id = r.resource_id
FROM fridges f
JOIN resources r
  ON r.kind='FRIDGE_COMPARTMENT' AND r.floor_no=f.floor_no AND r.display_order=c.position_order
WHERE c.fridge_id=f.fridge_id AND c.resource_id IS NULL;

-- 1:1 보장 제약 추가
ALTER TABLE compartments
  ALTER COLUMN resource_id SET NOT NULL;
ALTER TABLE compartments
  ADD CONSTRAINT fk_comp_resource FOREIGN KEY (resource_id) REFERENCES resources(resource_id);
ALTER TABLE compartments
  ADD CONSTRAINT uq_comp_resource UNIQUE(resource_id);

-- 라벨(묶음) + 개별 품목
CREATE TABLE item_groups (
  group_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  compartment_id BIGINT NOT NULL REFERENCES compartments(compartment_id),
  owner_user_id  BIGINT NOT NULL REFERENCES users(user_id),
  label_no       INTEGER NOT NULL CHECK (label_no > 0),           -- 스티커 번호
  status         TEXT NOT NULL DEFAULT 'ACTIVE'
                CHECK (status IN ('ACTIVE','REMOVED')),
  group_name     TEXT,                                            -- 묶음 이름
  group_note     TEXT,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  removed_at     TIMESTAMPTZ
);
-- 활성 라벨 유일성(삭제 즉시 재사용 허용)
CREATE UNIQUE INDEX ux_active_label_per_compartment
  ON item_groups(compartment_id, label_no) WHERE removed_at IS NULL;

CREATE TABLE item_units (
  item_id        BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  group_id       BIGINT NOT NULL REFERENCES item_groups(group_id) ON DELETE CASCADE,
  name           TEXT   NOT NULL,                                 -- 개별 품목명
  expiry_date    DATE,                                            -- 유통기한(NULL 허용)
  state          TEXT NOT NULL DEFAULT 'ACTIVE'
                 CHECK (state IN ('ACTIVE','REMOVED')),
  removed_reason TEXT CHECK (removed_reason IN ('CONSUMED','DISPOSED')),
  removed_at     TIMESTAMPTZ
);
CREATE INDEX ix_item_units_name_trgm ON item_units USING gin (name gin_trgm_ops);
CREATE INDEX ix_item_units_expiry    ON item_units(expiry_date);

-- 검사 세션(칸 잠금, 동시 2인), TTL은 앱이 ended_at 세팅으로 관리
CREATE TABLE inspection_sessions (
  session_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  compartment_id BIGINT  NOT NULL REFERENCES compartments(compartment_id),
  inspector_id   BIGINT  NOT NULL REFERENCES users(user_id),
  lock_slot      SMALLINT NOT NULL CHECK (lock_slot IN (1,2)),     -- 동시 2인 제어
  started_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  expires_at     TIMESTAMPTZ NOT NULL,                             -- 시작+5분 등
  ended_at       TIMESTAMPTZ
);
-- 활성 고유(시간 함수 사용 금지: 불변 파셜 인덱스)
CREATE UNIQUE INDEX ux_compartment_slot_active
  ON inspection_sessions(compartment_id, lock_slot) WHERE ended_at IS NULL;
CREATE UNIQUE INDEX ux_compartment_inspector_active
  ON inspection_sessions(compartment_id, inspector_id) WHERE ended_at IS NULL;

-- 검사 결과(사후 수정은 supersedes 체인으로)
CREATE TABLE inspection_actions (
  action_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id   BIGINT NOT NULL REFERENCES inspection_sessions(session_id) ON DELETE CASCADE,
  target_kind  TEXT   NOT NULL CHECK (target_kind IN ('GROUP','ITEM')),
  group_id     BIGINT REFERENCES item_groups(group_id),
  item_id      BIGINT REFERENCES item_units(item_id),
  action_type  TEXT   NOT NULL CHECK (action_type IN ('WARN','DISPOSE','NOTE')),
  reason       TEXT   CHECK (reason IN ('STICKER_MISSING','INFO_MISMATCH','EXPIRED','UNREGISTERED','STORAGE_ISSUE')),
  details      TEXT,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  supersedes_action_id BIGINT REFERENCES inspection_actions(action_id)
);
CREATE INDEX ix_inspect_actions_session ON inspection_actions(session_id);

-- =========================================================
-- 5) 알림 (유형 정의, 개인별 인스턴스, 읽음/전달 상태 + 설정)
-- =========================================================
CREATE TABLE notification_types (
  type_key    TEXT PRIMARY KEY,                                   -- 예: FRIDGE_EXPIRY_SOON
  category    TEXT NOT NULL CHECK (category IN ('SYSTEM','FRIDGE','LIBRARY','ROOM')),
  description TEXT
);
INSERT INTO notification_types(type_key,category,description) VALUES
  ('FRIDGE_EXPIRY_SOON','FRIDGE','유통기한 임박(1일 1회 묶음)'),
  ('FRIDGE_INSPECTION_RESULT','FRIDGE','정기점검 결과(경고/폐기)'),
  ('POINTS_CHANGED','SYSTEM','상벌점 변동'),
  ('LIB_DUE_SOON','LIBRARY','도서 반납 임박'),
  ('LIB_DUE_TODAY','LIBRARY','도서 반납 당일'),
  ('LIB_OVERDUE','LIBRARY','도서 연체'),
  ('LIB_HOLD_READY','LIBRARY','예약 도서 대출 가능'),
  ('LIB_HOLD_EXPIRED','LIBRARY','예약 만료')
ON CONFLICT DO NOTHING;

CREATE TABLE user_notifications (
  notif_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id      BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  type_key     TEXT   NOT NULL REFERENCES notification_types(type_key),
  payload      JSONB  NOT NULL,                                   -- {count, ids, deeplink,...}
  deeplink     TEXT,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  read_at      TIMESTAMPTZ,
  delivered_at TIMESTAMPTZ
);
CREATE INDEX ix_user_notifs_unread  ON user_notifications(user_id) WHERE read_at IS NULL;
CREATE INDEX ix_user_notifs_created ON user_notifications(user_id, created_at DESC);
CREATE INDEX ix_user_notifs_payload ON user_notifications USING gin(payload);

CREATE TABLE user_notification_settings (
  user_id               BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,
  push_opt_in           BOOLEAN NOT NULL DEFAULT TRUE,
  fridge_expiry_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,
  fridge_result_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,
  system_points_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,
  dnd_start             TIME,         -- 방해금지 시작(옵션)
  dnd_end               TIME
);

-- =========================================================
-- 6) 도서관 (메타/소장본/대출/예약/벌점 이벤트)
--     * 대기열 1명 정책. 반납 D-5/당일/연체 알림은 배치로 적재.
-- =========================================================
CREATE TABLE library_books (
  book_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  title       TEXT NOT NULL,
  author      TEXT,
  isbn13      TEXT UNIQUE,
  call_no     TEXT,                       -- 청구기호
  tags        TEXT[],
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at  TIMESTAMPTZ
);
CREATE INDEX ix_lib_books_title_trgm  ON library_books USING gin (title  gin_trgm_ops);
CREATE INDEX ix_lib_books_author_trgm ON library_books USING gin (author gin_trgm_ops);

CREATE TABLE library_copies (
  copy_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  book_id     BIGINT NOT NULL REFERENCES library_books(book_id) ON DELETE CASCADE,
  barcode     TEXT  NOT NULL UNIQUE,
  status      TEXT  NOT NULL DEFAULT 'AVAILABLE'
             CHECK (status IN ('AVAILABLE','LOANED','LOST','DISCARDED','MAINTENANCE')),
  location    TEXT  DEFAULT 'MAIN',
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE library_loans (
  loan_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  copy_id      BIGINT NOT NULL REFERENCES library_copies(copy_id),
  user_id      BIGINT NOT NULL REFERENCES users(user_id),
  loaned_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  due_date     DATE        NOT NULL,          -- 기본 14일 등은 앱에서 계산
  returned_at  TIMESTAMPTZ,
  extended_cnt SMALLINT    NOT NULL DEFAULT 0 CHECK (extended_cnt BETWEEN 0 AND 1),
  notes        TEXT
);
CREATE UNIQUE INDEX ux_lib_active_loan_per_copy
  ON library_loans(copy_id) WHERE returned_at IS NULL;
CREATE INDEX ix_lib_loans_user_active
  ON library_loans(user_id) WHERE returned_at IS NULL;

CREATE TABLE library_holds (
  hold_id         BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  book_id         BIGINT NOT NULL REFERENCES library_books(book_id),
  user_id         BIGINT NOT NULL REFERENCES users(user_id),
  status          TEXT   NOT NULL CHECK (status IN ('WAITING','READY','CANCELLED','EXPIRED','FULFILLED')),
  requested_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  ready_at        TIMESTAMPTZ,                 -- 픽업 준비
  pickup_deadline TIMESTAMPTZ,                 -- READY + n일
  cancelled_at    TIMESTAMPTZ,
  fulfilled_loan_id BIGINT REFERENCES library_loans(loan_id)
);
-- 책당 활성 예약 1건(대기열 1명)
CREATE UNIQUE INDEX ux_lib_book_active_hold
  ON library_holds(book_id) WHERE status IN ('WAITING','READY');
-- 사용자-책 중복 예약 방지
CREATE UNIQUE INDEX ux_lib_user_book_active_hold
  ON library_holds(book_id, user_id) WHERE status IN ('WAITING','READY');
CREATE INDEX ix_lib_holds_user_active
  ON library_holds(user_id) WHERE status IN ('WAITING','READY');

-- 연체 등 상벌점 이벤트(알림과 분리)
CREATE TABLE point_events (
  event_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id    BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  delta      INTEGER NOT NULL,                -- +상점 / -벌점
  reason     TEXT    NOT NULL CHECK (reason IN ('LIB_OVERDUE','MANUAL','OTHER')),
  meta       JSONB   NOT NULL DEFAULT '{}'::jsonb,  -- {loan_id, days_overdue}
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX ix_point_events_user ON point_events(user_id, created_at DESC);

COMMIT;
