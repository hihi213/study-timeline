## 6주 배포 MVP

- 포함: 냉장고 전 흐름, DB-락, SSE, 벌점, 최소 관리자, 배포.
- 제외: **Web Push**, Redis 분산락, Items v2, TRGM, 고급 대시보드.
- 컷리스트: Week4 통계 위젯→텍스트 수치. D-1 스케줄러→앱 진입 시 계산.

### **Week 1: 기반**

- 목표: 프로젝트·보안·DDL 골격 확정.
- 과업
    - Spring Boot 3(Java 21), Gradle, 멀티 프로필(dev/prod).
    - Flyway 적용 + **Fix Pack** 마이그(역할명 정합, resource_locks TTL=5m, event_log.actor_user_id/reason).
    - Security + JWT + RBAC(ADMIN_ROOT/ADMIN/INSPECTOR/RESIDENT).
    - CI 기초: 빌드·이미지 생성까지만.
- 게이트: 로그인 샘플 200/403, Flyway clean→migrate 무오류.

### **Week 2: 냉장고 등록·조회**

- 목표: 입주자 핵심 흐름 확보.
- 과업
    - 모델: **resources(FRIDGE_COMP)**, **item_bundles**만. SNNN= resource_id+label_no 런타임 생성.
    - API: 내 물품 등록(단일/묶음), 내 목록/상세, D-1 임박 조회.
    - 유일성: 활성 상태 (resource_id,label_no) 부분 UNIQUE.
    - event_log 적재(actor, reason 포함).
- 게이트: “등록→목록→상세” e2e, 100건 로컬 P95<150ms.

### **Week 3: 점검·리비전·벌점**

- 목표: 층별장 업무 플로우 완성.
- 과업

    - **DB 논리 락**: resource_locks + TTL 5분. 동시 2인 세션 제약.

### **Week 4: 실시간(SSE)·UX 최소**

- 목표: 포그라운드 실시간 반영.

- 게이트: 탭 2개 동시 갱신, 네트워크 단절 후 자동 재접속.

## 6주 DB

```sql
-- =========================================================
-- Dorm Convenience System — Final DDL (PostgreSQL ≥ 11)
-- 규칙
--   • ID: BIGINT IDENTITY
--   • 시간: timestamptz(UTC)
--   • 확장필드: attrs(JSONB)로 분기 처리
-- 라벨
--   • label_no 1..999만 DB가 강제
--   • 스티커 SNNN의 S(칸번호)는 앱이 계산. DB에는 NNN만 저장
-- 검사/정정
--   • 제출된 세션과 결과는 원본·정정 모두 불변
--   • 정정은 AMENDMENT 세션에서만 허용. correction 체인으로 연결
-- 조회
--   • v_inspection_results_effective로 정정 반영된 최종 결과만 노출
-- =========================================================
BEGIN;

-- 0) Extensions ---------------------------------------------------------
CREATE EXTENSION IF NOT EXISTS citext;  -- 대소문자 무시 계정/이메일

-- 1) Users / Roles ------------------------------------------------------
CREATE TABLE users (
  user_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  username      CITEXT NOT NULL UNIQUE,
  email         CITEXT UNIQUE,
  password_hash TEXT   NOT NULL,
  display_name  TEXT   NOT NULL,
  is_active     BOOLEAN NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE user_roles (
  user_id    BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  role       TEXT   NOT NULL CHECK (role IN ('ADMIN_ROOT','ADMIN','INSPECTOR','RESIDENT')),
  granted_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY(user_id, role)
);

-- 2) Resources ----------------------------------------------------------
CREATE TABLE resources (
  resource_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  type           TEXT NOT NULL,                     -- 예: FRIDGE_COMP, WASHER
  label          TEXT NOT NULL,                     -- 화면 표기용 명칭
  status         TEXT NOT NULL DEFAULT 'ACTIVE'
                 CHECK (status IN ('ACTIVE','OUT_OF_SERVICE')),
  floor_code     TEXT NOT NULL,                     -- 동/층 구분
  position_order SMALLINT,                          -- 칸 정렬 → S 계산에 사용
  comp_kind      TEXT,                              -- 예: FRIDGE/FREEZER
  attrs          JSONB NOT NULL DEFAULT '{}'::jsonb,-- 확장 속성
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE(type, label)
);
CREATE INDEX idx_resources_floor ON resources(floor_code);
CREATE INDEX idx_resources_attrs_gin ON resources USING GIN (attrs jsonb_path_ops);

-- 3) Items (Group / Unit) -----------------------------------------------
-- 그룹 = 스티커 한 장의 논리 단위. 라벨 번호는 "자원 내 활성 그룹"에서만 유일.
CREATE TABLE item_groups (
  group_id      BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  resource_id   BIGINT NOT NULL REFERENCES resources(resource_id) ON DELETE RESTRICT,
  owner_user_id BIGINT REFERENCES users(user_id) ON DELETE SET NULL,
  label_no      INTEGER NOT NULL CHECK (label_no BETWEEN 1 AND 999), -- NNN
  group_name    TEXT NOT NULL,                -- 대표명
  memo          TEXT,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  removed_at    TIMESTAMPTZ                    -- 소프트 종료. 번호 즉시 재사용 허용
);
-- 활성 그룹만 라벨 유일 보장 → 번호 재사용 정책 충족
CREATE UNIQUE INDEX ux_item_groups_label_active
  ON item_groups(resource_id, label_no) WHERE removed_at IS NULL;
CREATE INDEX idx_item_groups_active_owner
  ON item_groups(owner_user_id) WHERE removed_at IS NULL;
CREATE INDEX idx_item_groups_label_active_lookup
  ON item_groups(resource_id, label_no) WHERE removed_at IS NULL;

-- 유닛 = 그룹 내 개별 품목. 부분 소비/폐기에 대응해 유닛 단위로 제거 표시.
CREATE TABLE item_units (
  unit_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  group_id    BIGINT NOT NULL REFERENCES item_groups(group_id) ON DELETE CASCADE,
  name        TEXT NOT NULL,
  expires_on  DATE NOT NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  removed_at  TIMESTAMPTZ                      -- 부분/전량 제거 시각
);
CREATE INDEX idx_item_units_group   ON item_units(group_id);
CREATE INDEX idx_item_units_expires ON item_units(expires_on) WHERE removed_at IS NULL;
CREATE INDEX idx_item_units_group_active_exp
  ON item_units(group_id, expires_on)
  WHERE removed_at IS NULL INCLUDE (name);

-- [무결성#1] 종료 그룹에 유닛 추가/갱신 금지 -------------------------------
CREATE OR REPLACE FUNCTION ensure_unit_group_active()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE v_removed TIMESTAMPTZ;
BEGIN
  SELECT removed_at INTO v_removed FROM item_groups WHERE group_id = NEW.group_id;
  IF v_removed IS NOT NULL THEN
    RAISE EXCEPTION '종료된 그룹에는 유닛을 추가/갱신할 수 없습니다';
  END IF;
  RETURN NEW;
END$$;
CREATE TRIGGER trg_unit_group_active_ins BEFORE INSERT ON item_units
FOR EACH ROW EXECUTE FUNCTION ensure_unit_group_active();
CREATE TRIGGER trg_unit_group_active_upd BEFORE UPDATE ON item_units
FOR EACH ROW EXECUTE FUNCTION ensure_unit_group_active();

-- [편의] 활성 유닛이 0개가 되면 그룹 자동 종료 -----------------------------
CREATE OR REPLACE FUNCTION auto_close_group_when_empty()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE gid BIGINT;
BEGIN
  gid := COALESCE(NEW.group_id, OLD.group_id);
  IF NOT EXISTS (SELECT 1 FROM item_units WHERE group_id = gid AND removed_at IS NULL) THEN
    UPDATE item_groups SET removed_at = COALESCE(removed_at, now()) WHERE group_id = gid;
  END IF;
  RETURN COALESCE(NEW, OLD);
END$$;
CREATE TRIGGER trg_group_autoclose_ins  AFTER INSERT ON item_units  FOR EACH ROW EXECUTE FUNCTION auto_close_group_when_empty();
CREATE TRIGGER trg_group_autoclose_upd  AFTER UPDATE OF removed_at ON item_units FOR EACH ROW EXECUTE FUNCTION auto_close_group_when_empty();
CREATE TRIGGER trg_group_autoclose_del  AFTER DELETE ON item_units  FOR EACH ROW EXECUTE FUNCTION auto_close_group_when_empty();

-- [유틸] 그룹 종료 시 유닛 일괄 제거 타임스탬프 주입 -----------------------
CREATE OR REPLACE FUNCTION close_item_group(p_group_id BIGINT)
RETURNS VOID LANGUAGE plpgsql AS $$
BEGIN
  UPDATE item_groups SET removed_at = COALESCE(removed_at, now())
  WHERE group_id = p_group_id;
  UPDATE item_units SET removed_at = COALESCE(removed_at, now())
  WHERE group_id = p_group_id AND removed_at IS NULL;
END$$;

-- [유틸] 다음 사용 가능 라벨 추천 -----------------------------------------
CREATE OR REPLACE FUNCTION next_free_label(p_resource_id BIGINT)
RETURNS INTEGER LANGUAGE sql STABLE AS $$
WITH used AS (
  SELECT label_no FROM item_groups WHERE resource_id = p_resource_id AND removed_at IS NULL
),
seq AS (SELECT gs AS n FROM generate_series(1,999) gs)
SELECT MIN(n) FROM seq s LEFT JOIN used u ON u.label_no = s.n WHERE u.label_no IS NULL;
$$;

-- [유틸] 라벨 확정 + 그룹 생성(경합 안전) -----------------------------------
-- DB가 라벨 충돌을 최종적으로 차단. 프런트의 추천값은 힌트로만 사용.
CREATE OR REPLACE FUNCTION allocate_group_with_label(
  p_resource_id BIGINT,
  p_owner_user_id BIGINT,
  p_group_name TEXT,
  p_memo TEXT DEFAULT NULL
) RETURNS item_groups LANGUAGE plpgsql AS $$
DECLARE v_row item_groups;
BEGIN
  FOR i IN 1..15 LOOP
    WITH next AS (
      SELECT MIN(n) AS label_no
      FROM generate_series(1,999) AS s(n)
      LEFT JOIN item_groups g
        ON g.resource_id=p_resource_id AND g.removed_at IS NULL AND g.label_no=s.n
      WHERE g.label_no IS NULL
    )
    INSERT INTO item_groups(resource_id, owner_user_id, label_no, group_name, memo)
    SELECT p_resource_id, p_owner_user_id, label_no, p_group_name, p_memo FROM next
    WHERE label_no IS NOT NULL
    ON CONFLICT DO NOTHING
    RETURNING * INTO v_row;

    IF FOUND THEN RETURN v_row; END IF;
  END LOOP;
  RAISE EXCEPTION '사용 가능한 라벨이 없거나 경합이 과다합니다';
END$$;

-- 4) Inspection -------------------------------------------------------------
-- 검사 세션: 칸 단위 진행. IN_PROGRESS 단 하나만 허용(자원별).
-- 제출 시각을 확정하고 이후 불변. 정정은 별도 AMENDMENT 세션에서 수행.
CREATE TABLE inspection_sessions (
  session_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  resource_id   BIGINT NOT NULL REFERENCES resources(resource_id) ON DELETE RESTRICT,
  started_by    BIGINT NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
  status        TEXT  NOT NULL CHECK (status IN ('IN_PROGRESS','SUBMITTED','CANCELED')),
  started_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  ended_at      TIMESTAMPTZ,
  submitted_by  BIGINT REFERENCES users(user_id) ON DELETE SET NULL,
  submitted_at  TIMESTAMPTZ,
  -- 정정용 링크
  parent_session_id BIGINT REFERENCES inspection_sessions(session_id) ON DELETE SET NULL,
  purpose      TEXT NOT NULL DEFAULT 'NORMAL'
               CHECK (purpose IN ('NORMAL','AMENDMENT')),
  CONSTRAINT ck_is_amendment_parent
    CHECK ((purpose='AMENDMENT' AND parent_session_id IS NOT NULL)
        OR (purpose='NORMAL'    AND parent_session_id IS NULL))
);
CREATE UNIQUE INDEX ux_inspection_res_inprog
  ON inspection_sessions(resource_id) WHERE status='IN_PROGRESS';
CREATE INDEX idx_is_parent ON inspection_sessions(parent_session_id);

-- 세션 참여자: 동시 검사 기록 분리를 위한 명시적 참여 엔티티
CREATE TABLE inspection_participants (
  participant_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id     BIGINT NOT NULL REFERENCES inspection_sessions(session_id) ON DELETE CASCADE,
  user_id        BIGINT NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
  joined_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (session_id, user_id)
);
CREATE INDEX idx_ip_session ON inspection_participants(session_id);
Create INDEX idx_ip_user    ON inspection_participants(user_id);

-- 검사 결과: NO_STICKER는 아이템 참조 금지. 그 외에는 그룹/유닛 참조 필수.
CREATE TABLE inspection_results (
  result_id    BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_id   BIGINT NOT NULL REFERENCES inspection_sessions(session_id) ON DELETE CASCADE,
  group_id     BIGINT REFERENCES item_groups(group_id) ON DELETE SET NULL,
  unit_id      BIGINT REFERENCES item_units(unit_id) ON DELETE SET NULL,
  performer_participant_id BIGINT NOT NULL REFERENCES inspection_participants(participant_id) ON DELETE RESTRICT,
  action       TEXT NOT NULL CHECK (action IN ('PASS','WARNING','DISCARD','DISCARD_WITH_PENALTY')),
  violation_type TEXT CHECK (violation_type IS NULL OR violation_type IN ('EXPIRED','NO_STICKER','PACKAGING_DEFECT','INFO_MISMATCH')),
  note         TEXT,
  performed_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  -- 정정 체인
  correction_of_result_id BIGINT REFERENCES inspection_results(result_id) ON DELETE RESTRICT
);
CREATE INDEX idx_ir_session   ON inspection_results(session_id);
CREATE INDEX idx_ir_group     ON inspection_results(group_id);
CREATE INDEX idx_ir_unit      ON inspection_results(unit_id);
CREATE INDEX idx_ir_performer ON inspection_results(performer_participant_id);
CREATE INDEX idx_ir_time      ON inspection_results(performed_at DESC);
CREATE INDEX idx_ir_correction_of ON inspection_results(correction_of_result_id);

-- 세션 상태 전이 가드: 제출/취소 후 불변. 제출 시 submitted_at/ended_at 자동 확정.
CREATE OR REPLACE FUNCTION enforce_session_state_transitions()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF OLD.status IN ('SUBMITTED','CANCELED') AND NEW.status <> OLD.status THEN
    RAISE EXCEPTION '제출/취소된 세션은 상태 변경 불가';
  END IF;
  IF NEW.status='SUBMITTED' AND OLD.status <> 'SUBMITTED' THEN
    NEW.submitted_at := COALESCE(NEW.submitted_at, now());
    NEW.ended_at     := COALESCE(NEW.ended_at, now());
  END IF;
  IF NEW.status='IN_PROGRESS' AND (NEW.submitted_at IS NOT NULL OR NEW.ended_at IS NOT NULL) THEN
    RAISE EXCEPTION 'IN_PROGRESS는 submitted_at/ended_at이 NULL이어야 합니다';
  END IF;
  RETURN NEW;
END$$;
CREATE TRIGGER trg_is_state_transition
BEFORE UPDATE ON inspection_sessions
FOR EACH ROW EXECUTE FUNCTION enforce_session_state_transitions();

-- 결과-아이템-세션 자원 일치 가드 + NO_STICKER 규칙 강제
CREATE OR REPLACE FUNCTION ensure_result_item_refs()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE v_gid BIGINT; v_session_res BIGINT; v_item_res BIGINT;
BEGIN
  SELECT resource_id INTO v_session_res FROM inspection_sessions WHERE session_id = NEW.session_id;

  IF NEW.violation_type = 'NO_STICKER' THEN
    IF NEW.group_id IS NOT NULL OR NEW.unit_id IS NOT NULL THEN
      RAISE EXCEPTION 'NO_STICKER는 group_id/unit_id 없이 기록해야 합니다';
    END IF;
  ELSE
    IF NEW.group_id IS NULL AND NEW.unit_id IS NULL THEN
      RAISE EXCEPTION 'group_id 또는 unit_id 중 하나는 필수입니다';
    END IF;

    IF NEW.unit_id IS NOT NULL THEN
      SELECT group_id INTO v_gid FROM item_units WHERE unit_id = NEW.unit_id;
      IF NOT FOUND THEN RAISE EXCEPTION '존재하지 않는 unit_id입니다'; END IF;
      IF NEW.group_id IS NOT NULL AND NEW.group_id <> v_gid THEN
        RAISE EXCEPTION 'unit_id는 지정된 group_id에 속해야 합니다';
      END IF;
      SELECT resource_id INTO v_item_res FROM item_groups WHERE group_id = v_gid;
    ELSE
      SELECT resource_id INTO v_item_res FROM item_groups WHERE group_id = NEW.group_id;
      IF NOT FOUND THEN RAISE EXCEPTION '존재하지 않는 group_id입니다'; END IF;
    END IF;

    IF v_item_res IS DISTINCT FROM v_session_res THEN
      RAISE EXCEPTION '세션 리소스와 결과 리소스가 다릅니다';
    END IF;
  END IF;
  RETURN NEW;
END$$;
CREATE TRIGGER trg_ir_item_refs_ck
BEFORE INSERT OR UPDATE ON inspection_results
FOR EACH ROW EXECUTE FUNCTION ensure_result_item_refs();

-- 제출 후 결과 변경 차단(원본·정정 모두 불변) -------------------------------
CREATE OR REPLACE FUNCTION ensure_session_not_submitted()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE st TEXT;
BEGIN
  SELECT status INTO st FROM inspection_sessions WHERE session_id = COALESCE(NEW.session_id, OLD.session_id) FOR UPDATE;
  IF st='SUBMITTED' THEN
    RAISE EXCEPTION '이미 제출된 검사 세션입니다(결과 불변)';
  END IF;
  RETURN COALESCE(NEW, OLD);
END$$;
CREATE TRIGGER trg_ir_block_ins BEFORE INSERT ON inspection_results
FOR EACH ROW EXECUTE FUNCTION ensure_session_not_submitted();
CREATE TRIGGER trg_ir_block_upd BEFORE UPDATE ON inspection_results
FOR EACH ROW WHEN (OLD.* IS DISTINCT FROM NEW.*) EXECUTE FUNCTION ensure_session_not_submitted();
CREATE TRIGGER trg_ir_block_del BEFORE DELETE ON inspection_results
FOR EACH ROW EXECUTE FUNCTION ensure_session_not_submitted();

-- 정정 규칙: AMENDMENT 세션 한정. parent_session 일치, 대상은 SUBMITTED, 사유 필수.
CREATE OR REPLACE FUNCTION enforce_amendment_rules()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
  base_session_id BIGINT;
  base_resource   BIGINT;
  this_resource   BIGINT;
  base_status     TEXT;
  this_purpose    TEXT;
BEGIN
  IF NEW.correction_of_result_id IS NULL THEN
    RETURN NEW;
  END IF;

  SELECT purpose, resource_id INTO this_purpose, this_resource
  FROM inspection_sessions WHERE session_id = NEW.session_id;

  IF this_purpose <> 'AMENDMENT' THEN
    RAISE EXCEPTION '정정 결과는 AMENDMENT 세션에서만 생성할 수 있습니다';
  END IF;

  SELECT s.session_id, s.resource_id, s.status
    INTO base_session_id, base_resource, base_status
  FROM inspection_results r
  JOIN inspection_sessions s ON s.session_id = r.session_id
  WHERE r.result_id = NEW.correction_of_result_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION '존재하지 않는 correction_of_result_id';
  END IF;

  IF base_status <> 'SUBMITTED' THEN
    RAISE EXCEPTION '정정 대상 세션은 SUBMITTED 이어야 합니다';
  END IF;

  IF (SELECT parent_session_id FROM inspection_sessions WHERE session_id=NEW.session_id) <> base_session_id THEN
    RAISE EXCEPTION 'AMENDMENT 세션의 parent_session_id가 대상 세션과 일치해야 합니다';
  END IF;

  IF this_resource IS DISTINCT FROM base_resource THEN
    RAISE EXCEPTION '정정 세션 리소스와 대상 결과 리소스가 다릅니다';
  END IF;

  IF NEW.note IS NULL OR btrim(NEW.note) = '' THEN
    RAISE EXCEPTION '정정 사유(note)는 필수입니다';
  END IF;

  RETURN NEW;
END$$;
CREATE TRIGGER trg_ir_amend_rules
BEFORE INSERT ON inspection_results
FOR EACH ROW EXECUTE FUNCTION enforce_amendment_rules();

-- 최종 유효 결과 뷰: correction 체인의 leaf만 노출 -------------------------
CREATE OR REPLACE VIEW v_inspection_results_effective AS
SELECT r.*
FROM inspection_results r
WHERE NOT EXISTS (
  SELECT 1 FROM inspection_results c
  WHERE c.correction_of_result_id = r.result_id
);

-- 5) Penalties --------------------------------------------------------------
-- 벌점은 이벤트 로그성. ORIGINAL은 양수만, ADJUSTMENT는 가감조정.
CREATE TABLE penalties (
  penalty_id  BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id     BIGINT NOT NULL REFERENCES users(user_id) ON DELETE RESTRICT,
  inspection_result_id BIGINT REFERENCES inspection_results(result_id) ON DELETE SET NULL,
  source      TEXT NOT NULL CHECK (source IN ('ORIGINAL','ADJUSTMENT')),
  points      INTEGER NOT NULL,
  reason      TEXT NOT NULL,
  created_by  BIGINT REFERENCES users(user_id) ON DELETE SET NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  CHECK ((source='ORIGINAL' AND points>0) OR (source='ADJUSTMENT' AND points<>0))
);
CREATE INDEX idx_penalties_user ON penalties(user_id);
CREATE INDEX idx_penalties_user_time ON penalties(user_id, created_at DESC);

-- 사용자별 합계 뷰(원본/조정/총점)
CREATE OR REPLACE VIEW v_penalties_totals AS
SELECT u.user_id,
       COALESCE(SUM(p.points) FILTER (WHERE p.source='ORIGINAL'), 0)   AS original_points,
       COALESCE(SUM(p.points) FILTER (WHERE p.source='ADJUSTMENT'), 0) AS adjusted_points,
       COALESCE(SUM(p.points), 0)                                      AS total_points
FROM users u
LEFT JOIN penalties p ON p.user_id = u.user_id
GROUP BY u.user_id;

-- 6) Event Log --------------------------------------------------------------
-- 비즈니스 이벤트 감사. 파티셔닝 추천.
CREATE TABLE event_log (
  event_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  type           TEXT NOT NULL,            -- 예: items.created
  resource_id    BIGINT REFERENCES resources(resource_id) ON DELETE SET NULL,
  floor_code     TEXT,
  correlation_id TEXT,                     -- 요청/세션 상관 키
  data           JSONB NOT NULL,
  actor_user_id  BIGINT REFERENCES users(user_id),
  reason         TEXT
);
CREATE INDEX idx_eventlog_type_time ON event_log(type, created_at DESC);
-- 대량 이벤트 시 RANGE(created_at) 파티셔닝 권장

-- 7) 보조 유틸 --------------------------------------------------------------
-- 정정 세션 시작: 기준 세션은 SUBMITTED여야 하며 parent_session_id로 연결.
CREATE OR REPLACE FUNCTION start_amendment_session(p_base_session_id BIGINT, p_started_by BIGINT)
RETURNS inspection_sessions LANGUAGE plpgsql AS $$
DECLARE v inspection_sessions; DECLARE st TEXT;
BEGIN
  SELECT status INTO st FROM inspection_sessions WHERE session_id=p_base_session_id;
  IF st <> 'SUBMITTED' THEN
    RAISE EXCEPTION '기준 세션은 SUBMITTED 상태여야 합니다';
  END IF;

  INSERT INTO inspection_sessions(resource_id, started_by, status, purpose, parent_session_id, started_at)
  SELECT resource_id, p_started_by, 'IN_PROGRESS', 'AMENDMENT', p_base_session_id, now()
  FROM inspection_sessions WHERE session_id=p_base_session_id
  RETURNING * INTO v;

  RETURN v;
END$$;

COMMIT;
```

제공해주신 PostgreSQL DDL 스크립트를 기반으로 데이터베이스에 구현된 모든 시나리오를 기능 단위로 상세히 정리해 드리겠습니다.

## 1. 사용자 및 역할 관리 (RBAC) 🧑‍💼

- **사용자 생성 및 로그인**:
    - `users` 테이블에 사용자를 등록합니다. 아이디(`username`)와 이메일(`email`)은 **대소문자를 구분하지 않고** 유일해야 합니다 (`citext` 타입).
    - `password_hash` 필드는 해시된 비밀번호를 저장하여 보안을 유지합니다.
    - `is_active` 플래그를 통해 사용자를 비활성화(소프트 삭제)할 수 있습니다.
- **역할 부여 및 관리**:
    - `user_roles` 테이블을 통해 사용자에게 역할을 부여합니다.
    - 역할은 `ADMIN_ROOT`, `ADMIN`, `INSPECTOR`, `RESIDENT` 중 하나로 엄격하게 제한됩니다 (`CHECK` 제약조건). 이는 **역할 기반 접근 제어(RBAC)**의 핵심입니다.
    - 한 사용자는 여러 역할을 가질 수 있습니다 (예: 층별장이면서 입주자).

## 2. 물품(냉장고) 관리 🧊

이 시스템의 가장 복잡하고 핵심적인 부분으로, 물품 등록부터 폐기까지의 전체 생명주기를 관리합니다.

### 2.1. 물품 등록 및 라벨링

- **물품 묶음(그룹) 생성**:
    - 사용자가 냉장고 '칸'(`resources` 테이블, `type`='FRIDGE_COMP')을 선택하고 물품을 등록하면 `item_groups`에 하나의 레코드가 생성됩니다. 이것이 **스티커 한 장**에 해당하는 논리적 단위입니다.
    - **라벨 번호(NNN) 자동 할당**:
        - DB는 해당 칸에서 **사용 중이지 않은 가장 작은 번호**(1~999)를 찾아 `label_no`로 자동 할당합니다.
        - `allocate_group_with_label` 함수는 여러 사용자가 동시에 등록을 시도할 때 발생하는 **경합(Race Condition)을 방지**하며 안전하게 번호를 할당합니다.
        - 프론트엔드는 `next_free_label` 함수를 호출하여 사용자에게 추천 번호를 미리 보여줄 수 있습니다.
    - **라벨 번호 재사용**:
        - 물품 그룹이 제거되면(`removed_at` 기록) 해당 `label_no`는 즉시 다른 사용자가 재사용할 수 있습니다. 이는 `ux_item_groups_label_active`라는 **부분 고유 인덱스(Partial Unique Index)** 덕분에 가능합니다.
- **개별 물품(유닛) 관리**:
    - 하나의 물품 묶음(그룹)은 여러 개의 개별 물품(`item_units`)을 가질 수 있습니다.
    - 사용자는 묶음 내 일부 물품만 소비하거나 폐기할 수 있으며, 이는 `item_units`의 `removed_at`에 기록됩니다.

### 2.2. 물품 상태 관리 및 자동화

- **자동 그룹 종료**:
    - 한 그룹에 속한 **모든 유닛이 제거되면**(`removed_at`이 모두 채워지면), `auto_close_group_when_empty` 트리거가 자동으로 해당 그룹(`item_groups`)도 종료 처리합니다.
    - 이로 인해 관리자가 수동으로 빈 그룹을 정리할 필요가 없어집니다.
- **수동 그룹 종료**:
    - `close_item_group` 함수를 호출하면 그룹과 그에 속한 모든 활성 유닛을 한 번에 종료 처리할 수 있습니다.
- **데이터 무결성 유지**:
    - `ensure_unit_group_active` 트리거는 이미 **종료된 그룹에 새로운 유닛을 추가하거나 수정하는 것을 원천적으로 차단**하여 데이터 정합성을 보장합니다.

## 3. 정기 검사 및 결과 관리 🕵️

층별장(INSPECTOR)이 냉장고 칸을 검사하고 그 결과를 기록, 수정하는 전 과정을 체계적으로 관리합니다.

### 3.1. 검사 세션 진행

- **검사 시작**:
    - 층별장이 특정 칸(`resource_id`)에 대한 검사를 시작하면 `inspection_sessions`에 `status`가 'IN_PROGRESS'인 세션이 생성됩니다.
    - `ux_inspection_res_inprog` 고유 인덱스는 한 칸에 대해 **동시에 두 개 이상의 검사가 진행되는 것을 막습니다.** (시스템상의 잠금 역할)
- **동시 검사 지원**:
    - `inspection_participants` 테이블은 하나의 세션에 여러 검사자(최대 2명)가 참여할 수 있도록 설계되었습니다.
    - 누가 어떤 조치를 취했는지(`performer_participant_id`) 명확히 추적할 수 있습니다.

### 3.2. 검사 결과 기록 및 불변성

- **결과 기록**:
    - 검사자는 각 물품(또는 스티커 없는 물품)에 대해 `PASS`, `WARNING`, `DISCARD` 등의 조치를 `inspection_results`에 기록합니다.
- **결과의 불변성 (Immutability)**:
    - 세션이 'SUBMITTED' 상태로 변경되면, `ensure_session_not_submitted` 트리거가 해당 세션의 **모든 결과를 추가, 수정, 삭제하는 것을 완전히 차단**합니다. 이는 감사 추적을 위해 원본 기록을 훼손하지 않는 핵심 규칙입니다.
    - `enforce_session_state_transitions` 트리거는 제출된 세션의 상태를 다시 'IN_PROGRESS'로 되돌리는 등의 잘못된 상태 변경을 막습니다.

### 3.3. 검사 결과 정정 (수정)

- **정정 세션 시작**:
    - 이미 제출된 검사 결과를 수정해야 할 경우, 관리자는 `start_amendment_session` 함수를 호출하여 **정정 전용 세션('AMENDMENT' 목적)**을 새로 시작합니다.
    - 이 새 세션은 `parent_session_id`를 통해 원본 세션과 명확하게 연결됩니다.
- **정정 결과 기록**:
    - 정정 세션에서 새로운 결과를 기록할 때, `correction_of_result_id` 필드에 수정하려는 **원본 결과의 ID를 지정**합니다.
    - `enforce_amendment_rules` 트리거는 다음과 같은 엄격한 규칙을 강제합니다:
        - 정정은 'AMENDMENT' 세션에서만 가능합니다.
        - 정정 대상이 되는 원본 세션은 반드시 'SUBMITTED' 상태여야 합니다.
        - 정정 사유(`note`)를 반드시 기입해야 합니다.
- **최종 결과 조회**:
    - 사용자는 `v_inspection_results_effective` 뷰(VIEW)를 통해 복잡한 정정 기록에 상관없이 **항상 최신 유효 결과만** 볼 수 있습니다. 이 뷰는 정정 체인에서 가장 마지막에 있는 결과(leaf node)만 필터링하여 보여줍니다.

## 4. 벌점 시스템 📉

- **벌점 부과**:
    - 검사 결과(`inspection_result_id`)나 관리자의 직접 조치로 사용자에게 벌점을 부과하면 `penalties` 테이블에 기록됩니다.
    - `source` 필드는 이것이 최초 부과된 벌점(`ORIGINAL`)인지, 나중에 관리자가 조정한 것(`ADJUSTMENT`)인지를 구분합니다.
- **벌점 집계**:
    - `v_penalties_totals` 뷰는 사용자별로 최초 벌점, 조정된 점수, 그리고 **최종 합계 점수**를 항상 실시간으로 계산하여 제공합니다. 애플리케이션은 이 뷰를 조회하기만 하면 되므로 복잡한 계산 로직이 필요 없습니다.

## 5. 시스템 감사 및 유틸리티 🛠️

- **모든 활동 기록**:
    - `event_log` 테이블은 시스템에서 발생하는 주요 비즈니스 이벤트(예: 물품 생성, 검사 제출 등)를 기록하는 **감사 로그(Audit Log)** 역할을 합니다.
    - 누가(`actor_user_id`), 무엇을(`type`), 언제(`created_at`) 했는지 등을 추적할 수 있어 문제 발생 시 원인 파악에 용이합니다.
- **자원 관리**:
    - `resources` 테이블은 냉장고 칸, 세탁기 등 시스템이 관리하는 모든 물리적/논리적 자원을 정의합니다.
    - `position_order` 같은 필드는 스티커의 `S` 번호 계산 등 UI 표시 순서를 결정하는 데 사용됩니다.

### **입주자 (RESIDENT) 시나리오** 🧑‍🎓

입주자는 시스템의 가장 일반적인 사용자로, 주로 자신의 물품을 관리하는 기능과 상호작용합니다.

- **냉장고 물품 등록**
    - **동작**: 사용자가 냉장고의 특정 '칸'을 선택하고, 단일 또는 묶음 물품 정보를 입력하여 등록을 요청합니다.
    - **DB 시나리오**:
        
        1. 애플리케이션이 `next_free_label` 함수를 호출해 추천 라벨 번호(`NNN`)를 사용자에게 제시합니다.
        2. 사용자가 등록을 확정하면, `allocate_group_with_label` 함수가 호출됩니다. 이 함수는 경합 상태(Race Condition)를 방지하며 해당 칸(`resource_id`)에서 비어있는 `label_no`를 찾아 `item_groups` 테이블에 레코드를 안전하게 생성합니다.
        3. 생성된 `group_id`를 참조하여 개별 물품 정보(`이름`, `유통기한` 등)가 `item_units` 테이블에 저장됩니다.
            
- **자신의 물품 관리**
    - **동작**: 등록한 물품의 일부를 소비하거나, 묶음 전체를 폐기합니다.
    - **DB 시나리오**:
        - **부분 소비**: `item_units` 테이블에서 해당 물품의 `removed_at` 필드에 현재 시각을 기록합니다.
        - **전량 제거**: 그룹 내 모든 `item_units`의 `removed_at`을 기록합니다. 이때, `auto_close_group_when_empty` 트리거가 자동으로 `item_groups`의 `removed_at` 필드도 채워 그룹을 종료시킵니다.
        - **번호 재사용**: 그룹이 종료되는 즉시, 해당 `label_no`는 `ux_item_groups_label_active` 고유 인덱스 정책에 따라 다른 사용자가 즉시 재사용할 수 있게 됩니다.
- **검사 결과 및 벌점 확인**
    - **동작**: 정기 검사 후 자신의 물품이 경고 또는 폐기되었는지 확인하고, 부과된 벌점을 확인합니다.
    - **DB 시나리오**:
        - 애플리케이션은 `v_inspection_results_effective` 뷰를 조회하여 해당 사용자(`owner_user_id`)의 물품에 대한 최종 검사 결과를 보여줍니다.
        - `penalties` 테이블이나 집계 뷰인 `v_penalties_totals`를 조회하여 자신의 벌점 내역과 총점을 확인합니다.
- 칸 잘못등록했을시 칸이동 구현
	사용자가 물품을 다른 칸으로 옮길 때, 마치 그 칸에 새로 등록하는 것처럼 시스템이 **새로운 라벨 번호를 자동으로 부여**하는 방식입니다.
	
	- **동작 흐름**
	    
	    1. 사용자가 A물품(`group_id=101`, `resource_id=냉장1`, `label_no=123`)을 `냉동1` 칸으로 이동 요청합니다.
	    2. 시스템은 `냉동1` 칸에서 사용 가능한 다음 라벨 번호(예: `456`)를 `next_free_label` 함수나 유사 로직으로 찾습니다.
	    3. `item_groups` 테이블에서 `group_id=101`인 레코드의 `resource_id`를 `냉동1`으로, `label_no`를 `456`으로 **동시에 업데이트**합니다.
	    4. UI는 사용자에게 "물품이 냉동1-456으로 이동되었습니다. 스티커 번호를 수정해주세요." 라고 명확히 알려줍니다.

### **층별장 (INSPECTOR) 시나리오** 🕵️

층별장은 냉장고 정기 검사를 수행하는 핵심 권한을 가집니다. 데이터의 정확성과 기록의 무결성을 보장하는 강력한 규칙들이 적용됩니다.

- **냉장고 칸 점검 시작**
    - **동작**: 특정 칸의 검사를 시작하여 다른 사용자의 수정을 막는 '잠금' 상태로 만듭니다.
    - **DB 시나리오**:
        
        1. `inspection_sessions` 테이블에 `status`가 'IN_PROGRESS'인 세션이 생성됩니다.
        2. `ux_inspection_res_inprog` 고유 인덱스가 해당 `resource_id`에 대해 진행 중인 세션이 단 하나만 존재하도록 강제하여 동시 검사를 차단합니다.
        3. 검사를 시작한 층별장은 `inspection_participants` 테이블에 참여자로 기록되어 누가 검사를 수행했는지 추적할 수 있습니다.
            
- **검사 결과 기록**
    - **동작**: 각 물품에 대해 '통과', '경고', '폐기' 등의 조치를 기록하고, 스티커가 없는 미등록 물품도 처리합니다.
    - **DB 시나리오**:
        - 각 조치는 `inspection_results` 테이블에 한 줄의 레코드로 기록됩니다. `performer_participant_id`를 통해 어떤 검사자가 내린 조치인지 명확히 합니다.
        - `ensure_result_item_refs` 트리거는 데이터의 정합성을 검사합니다. 예를 들어, 'NO_STICKER' 위반 유형은 `group_id`나 `unit_id` 없이 기록되어야 함을 강제합니다.
- **검사 세션 제출 및 결과 확정**
    - **동작**: 검사를 완료하고 결과를 최종 제출합니다. 제출된 결과는 원칙적으로 수정할 수 없습니다.
    - **DB 시나리오**:
        - `inspection_sessions`의 `status`가 'SUBMITTED'로 변경됩니다. 이때 `enforce_session_state_transitions` 트리거가 `submitted_at`과 `ended_at` 타임스탬프를 자동으로 기록합니다.
        - 제출이 완료된 후, `ensure_session_not_submitted` 트리거가 해당 세션에 속한 모든 결과(`inspection_results`)에 대한 **추가, 수정, 삭제를 원천 차단**하여 기록의 **불변성(Immutability)**을 보장합니다.
- **제출된 결과 정정**
    - **동작**: 실수로 잘못 제출한 검사 결과를 사유와 함께 수정합니다.
    - **DB 시나리오**:
        
        1. `start_amendment_session` 함수를 호출하여 원본 세션을 부모로 하는 **'AMENDMENT' 목적의 새로운 정정 세션**을 시작합니다.
        2. 이 정정 세션 내에서 새로운 결과를 기록하되, `correction_of_result_id` 필드에 **수정 대상인 원본 결과의 ID**를 명시합니다.
        3. `enforce_amendment_rules` 트리거가 '정정은 AMENDMENT 세션에서만 가능', '원본 세션은 SUBMITTED 상태여야 함', '정정 사유(`note`) 필수' 등의 모든 규칙을 강제합니다.
        4. 일반 조회 시에는 `v_inspection_results_effective` 뷰를 사용하여 복잡한 정정 기록과 상관없이 항상 최종 유효 결과만 노출됩니다.

### **관리자 (ADMIN) 시나리오** 🛠️

관리자는 시스템의 자원, 사용자 권한, 정책을 관리하고 예외 상황을 처리하는 포괄적인 권한을 가집니다.

- **자원 및 사용자 관리**
    - **동작**: 냉장고 칸이나 세탁기 등의 자원을 추가/수정하고, 특정 입주자를 층별장으로 임명하거나 해제합니다.
    - **DB 시나리오**:
        - `resources` 테이블의 데이터를 직접 추가하거나 `status`를 'OUT_OF_SERVICE'로 변경하여 자원을 관리합니다.
        - `user_roles` 테이블에 레코드를 추가하거나 삭제하여 사용자에게 'INSPECTOR' 역할을 부여하거나 회수합니다.
        - `users` 테이블의 `is_active` 플래그를 `FALSE`로 변경하여 특정 사용자의 계정을 비활성화할 수 있습니다.
- **벌점 조정**
    - **동작**: 시스템에 의해 자동 부과된 벌점을 검토하고, 필요시 수동으로 가감합니다.
    - **DB 시나리오**:
        - `penalties` 테이블에 `source`를 'ADJUSTMENT'로 하여 새로운 레코드를 추가합니다. `points` 값은 양수 또는 음수가 될 수 있어 벌점을 가감하는 것이 모두 가능합니다.
        - `v_penalties_totals` 뷰를 통해 모든 사용자의 현재 총점을 손쉽게 파악합니다.
- **시스템 전체 이벤트 감사**
    - **동작**: 시스템에서 발생한 모든 중요한 활동 이력을 추적하여 문제 발생 시 원인을 파악합니다.
    - **DB 시나리오**: `event_log` 테이블을 조회하여 누가(`actor_user_id`), 언제(`created_at`), 어떤 리소스에 대해(`resource_id`), 무슨 작업(`type`)을 했는지 상세히 확인할 수 있습니다. 이는 시스템의 모든 활동에 대한 감사 로그 역할을 합니다.
