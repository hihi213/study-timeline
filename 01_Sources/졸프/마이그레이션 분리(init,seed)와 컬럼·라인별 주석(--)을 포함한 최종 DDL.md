CREATE TABLE user_room_assignments (                                   -- 사용자 호실+개인번호 할당(이력)

ALTER TABLE user_room_assignments

  inspector_id BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE, -- 검사자 FK

  lock_slot    SMALLINT NOT NULL,                                       -- 1 또는 2
  state        TEXT NOT NULL DEFAULT 'ACTIVE',                          -- ACTIVE/CLOSED
  created_at   timestamptz NOT NULL DEFAULT now(),                      -- 생성시각

  ended_at     timestamptz                                              -- 종료 시각
);

  ADD CONSTRAINT chk_session_state CHECK (                               -- 상태-시간 일관

  ON inspection_sessions(floor_no, lock_slot) WHERE state = 'ACTIVE';

CREATE TABLE inspection_actions (                                       -- 검사 조치 기록
  action_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,    -- PK
  session_id    BIGINT NOT NULL REFERENCES inspection_sessions(session_id) ON DELETE CASCADE, -- 세션 FK
  action_type   TEXT   NOT NULL,                                        -- 'WARN' | 'DISPOSE'
  reason        TEXT   NOT NULL,                                        -- INFO_MISMATCH/STORAGE_ISSUE/STICKER_MISSING_REGISTERED/EXPIRED/UNREGISTERED
  target_kind   TEXT   NOT NULL DEFAULT 'ITEM_OR_GROUP',                -- ITEM_OR_GROUP | UNREGISTERED
  group_id      BIGINT REFERENCES item_groups(group_id) ON DELETE SET NULL,  -- 그룹 대상(옵션)
  item_id       BIGINT REFERENCES item_units(item_id)   ON DELETE SET NULL,  -- 품목 대상(옵션)
  compartment_id BIGINT REFERENCES compartments(compartment_id) ON DELETE SET NULL, -- 미등록용 칸 지정
  evidence      JSONB  NOT NULL DEFAULT '{}'::jsonb,                    -- 증빙(사진/메모)
  details       TEXT,                                                   -- 자유 코멘트
  created_at    timestamptz NOT NULL DEFAULT now()                      -- 생성시각
);

  );

CREATE OR REPLACE FUNCTION fn_compute_item_group_shortcode() RETURNS trigger AS $$ -- 그룹 short_code
DECLARE v_slot smallint;
BEGIN
  SELECT c.slot_no INTO v_slot FROM compartments c WHERE c.compartment_id = NEW.compartment_id;
  NEW.short_code := COALESCE(v_slot::text,'X') || LPAD(NEW.label_no::text,3,'0');
  RETURN NEW;
END; $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_item_groups_shortcode ON item_groups;
CREATE TRIGGER trg_item_groups_shortcode
BEFORE INSERT OR UPDATE OF compartment_id, label_no ON item_groups
FOR EACH ROW EXECUTE FUNCTION fn_compute_item_group_shortcode();

CREATE OR REPLACE FUNCTION fn_on_inspection_action() RETURNS trigger AS $$ -- 조치→물품 상태 반영
BEGIN
  IF NEW.action_type = 'DISPOSE' THEN
    IF NEW.item_id IS NOT NULL THEN
      UPDATE item_units
         SET status='REMOVED', removed_reason=NEW.reason, removed_at=now()
       WHERE item_id = NEW.item_id AND status <> 'REMOVED';
    ELSIF NEW.group_id IS NOT NULL THEN
      UPDATE item_units
         SET status='REMOVED', removed_reason=NEW.reason, removed_at=now()
       WHERE group_id = NEW.group_id AND status <> 'REMOVED';
      UPDATE item_groups
         SET status='REMOVED', removed_reason=NEW.reason, removed_at=now()
       WHERE group_id = NEW.group_id AND status <> 'REMOVED';
    END IF;
  END IF;
  RETURN NEW;
END; $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_inspection_action_apply ON inspection_actions;
CREATE TRIGGER trg_inspection_action_apply AFTER INSERT ON inspection_actions
FOR EACH ROW EXECUTE FUNCTION fn_on_inspection_action();

CREATE OR REPLACE FUNCTION fn_on_penalty_insert() RETURNS trigger AS $$  -- 상벌점→알림
BEGIN
  INSERT INTO user_notifications(user_id, type_code, title, body, payload, sent_at)
  SELECT NEW.user_id, 'POINTS_CHANGED',
         '상벌점 변동',
         CASE WHEN NEW.delta_points>=0 THEN ('+'||NEW.delta_points::text) ELSE NEW.delta_points::text END,
         jsonb_build_object('delta', NEW.delta_points, 'reason', NEW.reason, 'penalty_id', NEW.penalty_id),
         now()
  WHERE EXISTS (SELECT 1 FROM notification_types WHERE type_code='POINTS_CHANGED');
  RETURN NEW;
END; $$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION fn_apply_compartment_status_event() RETURNS trigger AS $$ -- 상태이력→스냅샷+로그
DECLARE v_prev TEXT;
BEGIN
  SELECT status INTO v_prev FROM compartments WHERE compartment_id = NEW.compartment_id;
  UPDATE compartments SET status = NEW.new_status WHERE compartment_id = NEW.compartment_id;
  INSERT INTO event_log(actor_user_id, action, entity_kind, entity_id, message, meta)
  VALUES (NEW.actor_user_id, 'RESOURCE_STATUS_CHANGED', 'COMPARTMENT', NEW.compartment_id,
          'compartment status changed',
          jsonb_build_object('old', v_prev, 'new', NEW.new_status, 'reason', NEW.reason, 'event_id', NEW.event_id));
  RETURN NEW;
END; $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_rse_apply ON resource_status_events;
CREATE TRIGGER trg_rse_apply AFTER INSERT ON resource_status_events
FOR EACH ROW EXECUTE FUNCTION fn_apply_compartment_status_event();

CREATE OR REPLACE FUNCTION close_expired_sessions() RETURNS integer AS $$ -- 세션 만료 처리(크론 호출)
DECLARE v_cnt integer;
BEGIN
  UPDATE inspection_sessions
     SET state='CLOSED', ended_at=now()
   WHERE state='ACTIVE' AND expires_at < now();
  GET DIAGNOSTICS v_cnt = ROW_COUNT;
  RETURN v_cnt;
END; $$ LANGUAGE plpgsql;
