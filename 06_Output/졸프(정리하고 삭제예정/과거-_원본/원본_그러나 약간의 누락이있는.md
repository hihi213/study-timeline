```sql
-- 스냅샷 초기화 트리거(자원 생성 시 AVAILABLE)
CREATE OR REPLACE FUNCTION trg_resources_init_status() RETURNS trigger AS $$
BEGIN
  INSERT INTO resource_status(resource_id, status, since)
  VALUES (NEW.resource_id, 'AVAILABLE', now())
  ON CONFLICT (resource_id) DO NOTHING;
  RETURN NEW;
END $$ LANGUAGE plpgsql;

AFTER INSERT ON resources
FOR EACH ROW EXECUTE FUNCTION trg_resources_init_status();

-- 이력→스냅샷 동기화 트리거
CREATE OR REPLACE FUNCTION trg_rse_apply_snapshot() RETURNS trigger AS $$
BEGIN
  INSERT INTO resource_status(resource_id, status, since)
  VALUES (NEW.resource_id, NEW.to_status, NEW.created_at)
  ON CONFLICT (resource_id)
  DO UPDATE SET status = EXCLUDED.status, since = EXCLUDED.since;
  RETURN NEW;
END $$ LANGUAGE plpgsql;

-- 칸 INSERT → 자원 자동 생성
CREATE OR REPLACE FUNCTION compartment_autoresource_ins() RETURNS trigger AS $$
DECLARE v_floor smallint; v_name text; v_res bigint;
BEGIN
  SELECT floor_no INTO v_floor FROM fridges WHERE fridge_id = NEW.fridge_id; -- 층 조회
  v_name := COALESCE(NEW.name,
            CASE WHEN NEW.kind='FRIDGE' THEN '냉장' || NEW.position_order ELSE '냉동' END); -- 기본명

  INSERT INTO resources(kind, name, floor_no, display_order, access_scope, attrs)
  VALUES ('FRIDGE_COMPARTMENT', v_name, v_floor, NEW.position_order, 'ALL', '{}'::jsonb)
  RETURNING resource_id INTO v_res;                        -- 자원 생성

  NEW.resource_id := v_res;                                -- 자원 FK 연결
  RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_compartment_autoresource_ins ON compartments;
CREATE TRIGGER trg_compartment_autoresource_ins
BEFORE INSERT ON compartments
FOR EACH ROW
WHEN (NEW.resource_id IS NULL)
EXECUTE FUNCTION compartment_autoresource_ins();

-- 칸 정보 변경 → 자원 동기화
CREATE OR REPLACE FUNCTION compartment_autoresource_upd() RETURNS trigger AS $$
DECLARE v_floor smallint; v_name text;
BEGIN
  SELECT floor_no INTO v_floor FROM fridges WHERE fridge_id = NEW.fridge_id; -- 층 조회
  v_name := COALESCE(NEW.name,
            CASE WHEN NEW.kind='FRIDGE' THEN '냉장' || NEW.position_order ELSE '냉동' END); -- 기본명

  UPDATE resources
     SET name = v_name,
         floor_no = v_floor,
         display_order = NEW.position_order
   WHERE resource_id = NEW.resource_id;                   -- 자원 스냅샷 반영

  RETURN NEW;
END $$ LANGUAGE plpgsql;

AFTER UPDATE OF name, position_order, kind ON compartments
FOR EACH ROW
WHEN (NEW.resource_id IS NOT NULL)
EXECUTE FUNCTION compartment_autoresource_upd();

-- 단축코드 계산 함수(칸 position_order + label_no)
CREATE OR REPLACE FUNCTION fn_compute_shortcode(p_compartment_id BIGINT, p_label INT)
RETURNS TEXT LANGUAGE plpgsql STABLE AS $$
DECLARE v_pos SMALLINT;
BEGIN
  SELECT position_order INTO v_pos FROM compartments WHERE compartment_id = p_compartment_id; -- 칸 순서
  IF v_pos IS NULL THEN RETURN NULL; END IF;
  RETURN v_pos::text || lpad(p_label::text, 3, '0'); -- 예: 1 + 023 = '1023'
END $$;

-- 라벨 INSERT/UPDATE 시 short_code 세팅
CREATE OR REPLACE FUNCTION trg_ig_set_shortcode() RETURNS trigger AS $$
BEGIN
  NEW.short_code := fn_compute_shortcode(NEW.compartment_id, NEW.label_no);
  RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS ig_set_shortcode ON item_groups;

BEFORE INSERT OR UPDATE OF compartment_id, label_no ON item_groups
FOR EACH ROW EXECUTE FUNCTION trg_ig_set_shortcode();

-- 칸 position 변경 시 관련 short_code 일괄 갱신
CREATE OR REPLACE FUNCTION trg_comp_update_shortcode() RETURNS trigger AS $$
BEGIN
  IF NEW.position_order <> OLD.position_order THEN
    UPDATE item_groups ig
       SET short_code = fn_compute_shortcode(ig.compartment_id, ig.label_no)
     WHERE ig.compartment_id = NEW.compartment_id;
  END IF;
  RETURN NEW;
END $$ LANGUAGE plpgsql;

-- 층별장/ADMIN 권한 검증 트리거
CREATE OR REPLACE FUNCTION trg_inspection_auth() RETURNS trigger AS $$
DECLARE v_floor smallint; v_is_head int; v_is_admin int;
BEGIN
  SELECT f.floor_no INTO v_floor
  FROM compartments c JOIN fridges f ON f.fridge_id = c.fridge_id
  WHERE c.compartment_id = NEW.compartment_id;                -- 칸의 층

  SELECT COUNT(*) INTO v_is_head
  FROM floor_head_assignments
  WHERE floor_no = v_floor AND user_id = NEW.inspector_id AND revoked_at IS NULL; -- 층별장 여부

  SELECT COUNT(*) INTO v_is_admin
  FROM user_roles ur JOIN roles r ON r.role_id = ur.role_id
  WHERE ur.user_id = NEW.inspector_id AND r.name IN ('ADMIN','ADMIN_ROOT');       -- 관리자 여부

  IF v_is_head = 0 AND v_is_admin = 0 THEN
    RAISE EXCEPTION 'inspector % is not authorized for floor %', NEW.inspector_id, v_floor;
  END IF;
  RETURN NEW;
END $$ LANGUAGE plpgsql;

  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),        -- 생성시각
  read_at      TIMESTAMPTZ,                               -- 읽음시각
  delivered_at TIMESTAMPTZ,                               -- 최초 전달시각(요약)
  created_on   DATE GENERATED ALWAYS AS (created_at::date) STORED -- 생성일(중복억제)
);
CREATE INDEX ix_user_notifs_unread
  ON user_notifications(user_id) WHERE read_at IS NULL;   -- 미읽음 조회

CREATE UNIQUE INDEX ux_un_fridge_expiry_daily
  ON user_notifications(user_id, created_on)
  WHERE type_key = 'FRIDGE_EXPIRY_SOON';                  -- 하루 1회 묶음

CREATE TABLE user_notification_deliveries (               -- 알림 전달 로그(엔드포인트별)
  delivery_id  BIGSERIAL PRIMARY KEY,                     -- PK
  notif_id     BIGINT NOT NULL REFERENCES user_notifications(notif_id) ON DELETE CASCADE, -- 알림 FK
  endpoint_id  BIGINT NOT NULL REFERENCES push_endpoints(endpoint_id) ON DELETE CASCADE,   -- 엔드포인트 FK
  attempt_cnt  INT NOT NULL DEFAULT 0,                    -- 시도 횟수
  last_status  TEXT,                                      -- 'SENT','FAILED' 등
  last_error   TEXT,                                      -- 에러 메시지
  updated_at   TIMESTAMPTZ NOT NULL DEFAULT now(),        -- 갱신 시각
  UNIQUE (notif_id, endpoint_id)                          -- 중복 방지
);
CREATE INDEX ix_notif_deliveries_notif
  ON user_notification_deliveries(notif_id);              -- 알림별 전달 로그

CREATE TABLE user_notification_settings (                 -- 사용자 알림 설정
  user_id               BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE, -- 사용자 FK
  push_opt_in           BOOLEAN NOT NULL DEFAULT TRUE,    -- 전체 푸시 동의
  fridge_expiry_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,    -- 냉장 임박 동의
  fridge_result_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,    -- 점검 결과 동의
  system_points_opt_in  BOOLEAN NOT NULL DEFAULT TRUE,    -- 상벌점 알림 동의
  dnd_start             TIME,                              -- 방해금지 시작
  dnd_end               TIME                               -- 방해금지 종료
);
ALTER TABLE user_notification_settings
  ADD CONSTRAINT chk_dnd_window CHECK (
    dnd_start IS NULL OR dnd_end IS NULL OR dnd_start < dnd_end   -- 단순 유효성
);

-- 사용자 생성 시 기본 알림 설정 자동 생성
CREATE OR REPLACE FUNCTION trg_user_settings_ins() RETURNS trigger AS $$
BEGIN
  INSERT INTO user_notification_settings(user_id) VALUES (NEW.user_id)
  ON CONFLICT (user_id) DO NOTHING;
  RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS users_init_settings ON users;
CREATE TRIGGER users_init_settings
AFTER INSERT ON users
FOR EACH ROW EXECUTE FUNCTION trg_user_settings_ins();
```

```sql
BEGIN;

-- [RBAC] 기본 역할 시드
INSERT INTO roles(role_id,name) VALUES
 (0,'ADMIN_ROOT'),(1,'ADMIN'),(2,'INSPECTOR'),(3,'RESIDENT')
ON CONFLICT DO NOTHING;  -- 중복 방지

-- [ORG] 층 시드
INSERT INTO floors(floor_no,gender) VALUES
 (2,'M'),(3,'F'),(4,'F'),(5,'F')
ON CONFLICT DO NOTHING;  -- 중복 방지

-- [FRIDGE] 층별 냉장고 시드
INSERT INTO fridges(floor_no) VALUES (2),(3),(4),(5)
ON CONFLICT DO NOTHING;  -- 중복 방지

-- [FRIDGE] 칸 시드(각 층: 냉장×3 + 냉동×1)
INSERT INTO compartments (fridge_id, position_order, kind, name)
SELECT fr.fridge_id, pos,                               -- 칸 순번
       CASE WHEN pos <= 3 THEN 'FRIDGE' ELSE 'FREEZER' END, -- 유형
       CASE WHEN pos <= 3 THEN CONCAT('냉장',pos) ELSE '냉동' END -- 기본명
FROM fridges fr
CROSS JOIN generate_series(1,4) AS pos
ON CONFLICT DO NOTHING;                                 -- 중복 방지
-- ↑ 자원(resources)·상태 스냅샷은 트리거로 자동 생성됨

-- [NOTIF] 알림 유형 시드
INSERT INTO notification_types(type_key,category,description) VALUES
  ('FRIDGE_EXPIRY_SOON','FRIDGE','유통기한 임박(1일 1회 묶음)'),
  ('FRIDGE_INSPECTION_RESULT','FRIDGE','정기점검 결과(경고/폐기)'),
  ('POINTS_CHANGED','SYSTEM','상벌점 변동')
ON CONFLICT DO NOTHING;                                 -- 중복 방지

COMMIT;
```

- 검사 세션 만료 처리는 스케줄러에서 ended_at IS NULL AND expires_at < now() 종료.
