권장패치를 반영한 최초로 적용할(drop등이 없는) 초기화(init)와 시드(seed) 코드를 생성해라.이때 두 코드 모두 한줄의 단위가 73자임을 감안하여 가독성을 고려하여 작성하며(예시, 만일 빈곳이 아닌 코드옆에 주석을 달시 한칸의 여백만 유지한다), 73글자를 초과할시 주석 전체를 다음줄이나 이전줄에다가 작성한다 주석은 코드에 대한 설명으로, 처음 DB를 접하고 배우는 사람이 이해할수있도록 코드의 기능 단위로 작성한다.
```sql
/* ======================================================================
   V1__init_mvp.sql
   PostgreSQL 14+
   목적: 사용자·공간·자원·검사/조치·RBAC·알림·상벌점·상태이력·감사로그
   최초 적용용 DDL (DROP/REPLACE 미사용)
   ====================================================================== */
BEGIN; -- 트랜잭션 시작

/* ============================== 확장 ================================== */

CREATE TABLE roles ( -- 역할 마스터
  role_id     BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, -- PK
  code        TEXT UNIQUE NOT NULL, -- ADMIN_ROOT/ADMIN/...
  name        TEXT NOT NULL, -- 역할명
  created_at  timestamptz NOT NULL DEFAULT now() -- 생성시각
); -- roles

CREATE TABLE users ( -- 사용자 계정
  user_id       BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, -- PK
  username      TEXT UNIQUE NOT NULL, -- 로그인 ID
  display_name  TEXT NOT NULL, -- 표시 이름
  email         CITEXT UNIQUE, -- 이메일(선택)
  password_hash TEXT NOT NULL, -- 비밀번호 해시
  gender        TEXT NOT NULL, -- MEN/WOMEN
  phone         TEXT, -- 연락처(선택)
  is_active     BOOLEAN NOT NULL DEFAULT TRUE, -- 활성 여부
  created_at    timestamptz NOT NULL DEFAULT now() -- 생성시각
); -- users
ALTER TABLE users
  ADD CONSTRAINT chk_users_gender CHECK (gender IN ('MEN','WOMEN')); -- 성별 제한

CREATE TABLE user_roles ( -- 사용자-역할 매핑
  user_id  BIGINT REFERENCES users(user_id) ON DELETE CASCADE, -- 사용자 FK
  role_id  BIGINT REFERENCES roles(role_id) ON DELETE CASCADE, -- 역할 FK
  PRIMARY KEY (user_id, role_id) -- 복합 PK
); -- user_roles

               STORED, -- 201 등

  capacity     SMALLINT NOT NULL DEFAULT 3, -- 수용 인원
  occupied     SMALLINT NOT NULL DEFAULT 0, -- 현재 점유 인원

ALTER TABLE rooms
  ADD CONSTRAINT chk_room_type CHECK (room_type IN ('TRIPLE','SINGLE')); -- 타입
ALTER TABLE rooms
  ADD CONSTRAINT chk_capacity CHECK (capacity BETWEEN 1 AND 4); -- 용량
ALTER TABLE rooms

CREATE TABLE user_room_assignments ( -- 사용자 호실 배정(이력)
  user_id     BIGINT NOT NULL REFERENCES users(user_id) ON DELETE CASCADE, -- FK
  room_id     BIGINT NOT NULL REFERENCES rooms(room_id) ON DELETE RESTRICT, -- FK
  personal_no SMALLINT NOT NULL, -- 개인 번호 1..3
  assigned_at timestamptz NOT NULL DEFAULT now(), -- 배정 시각
  revoked_at  timestamptz, -- 회수 시각(NULL=활성)
  PRIMARY KEY (user_id, assigned_at) -- 이력형 PK
); -- user_room_assignments
ALTER TABLE user_room_assignments
  ADD CONSTRAINT chk_personal_no CHECK (personal_no BETWEEN 1 AND 3); -- 범위
ALTER TABLE user_room_assignments
  ADD CONSTRAINT chk_assignment_time CHECK
  (revoked_at IS NULL OR revoked_at > assigned_at); -- 시간 일관
CREATE UNIQUE INDEX uq_user_active_assignment
  ON user_room_assignments(user_id) WHERE revoked_at IS NULL; -- 활성 1건
CREATE INDEX ix_assignment_room
  ON user_room_assignments(room_id) WHERE revoked_at IS NULL; -- 방 조회
CREATE UNIQUE INDEX uq_active_personal_on_room
  ON user_room_assignments(room_id, personal_no)
  WHERE revoked_at IS NULL; -- 방 내 개인번호 유니크
CREATE INDEX ix_ura_user_active
  ON user_room_assignments(user_id, assigned_at DESC)
  WHERE revoked_at IS NULL; -- 활성 탐색

CREATE VIEW v_user_active_assignment AS -- 활성 사용자 배정 뷰
SELECT a.user_id, a.room_id, r.floor_no, a.personal_no
FROM user_room_assignments a
JOIN rooms r ON r.room_id=a.room_id
WHERE a.revoked_at IS NULL; -- 활성만

-- 그룹 short_code 계산(FREEZER='F', 그 외 slot_no)
CREATE FUNCTION fn_compute_item_group_shortcode()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_slot smallint; v_kind text;
BEGIN
  SELECT c.slot_no, c.attrs->>'kind'
    INTO v_slot, v_kind
  FROM compartments c
  WHERE c.compartment_id=NEW.compartment_id; -- 칸 속성
  IF v_kind IS NULL THEN
    RAISE EXCEPTION 'compartment % has no attrs.kind', NEW.compartment_id;
  END IF;
  NEW.short_code :=
    CASE WHEN v_kind='FREEZER' THEN 'F' ELSE v_slot::text END
    || LPAD(NEW.label_no::text,3,'0'); -- 코드 생성
  RETURN NEW;
END; $$; -- 함수
CREATE TRIGGER trg_item_groups_shortcode

-- 검사 조치→물품 상태 반영(DISPOSE 시 제거)
CREATE FUNCTION fn_on_inspection_action()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.action_type='DISPOSE' THEN
    IF NEW.item_id IS NOT NULL THEN
      UPDATE item_units
         SET status='REMOVED',
             removed_reason=NEW.reason,
             removed_at=now()
       WHERE item_id=NEW.item_id AND status<>'REMOVED';
    ELSIF NEW.group_id IS NOT NULL THEN
      UPDATE item_units
         SET status='REMOVED',
             removed_reason=NEW.reason,
             removed_at=now()
       WHERE group_id=NEW.group_id AND status<>'REMOVED';
      UPDATE item_groups
         SET status='REMOVED',
             removed_reason=NEW.reason,
             removed_at=now()
       WHERE group_id=NEW.group_id AND status<>'REMOVED';
    END IF;
  END IF;
  RETURN NEW;
END; $$; -- 함수

-- 상태이력 old_status 자동 채움·검증
CREATE FUNCTION tg_rse_fill_old()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_curr text;
BEGIN
  SELECT status INTO v_curr
  FROM compartments WHERE compartment_id=NEW.compartment_id; -- 현재 상태
  IF v_curr IS NULL THEN
    RAISE EXCEPTION 'compartment % not found', NEW.compartment_id;
  END IF;
  IF NEW.old_status IS NULL THEN
    NEW.old_status := v_curr; -- 자동 채움
  ELSIF NEW.old_status <> v_curr THEN
    RAISE EXCEPTION 'old_status % mismatches current %',
      NEW.old_status, v_curr;
  END IF;
  IF NEW.old_status = NEW.new_status THEN
    RAISE EXCEPTION 'no-op status change % -> %',
      NEW.old_status, NEW.new_status;
  END IF;
  RETURN NEW;
END; $$; -- 함수
CREATE TRIGGER trg_rse_fill_old
BEFORE INSERT ON resource_status_events
FOR EACH ROW EXECUTE FUNCTION tg_rse_fill_old(); -- 보정

RETURNS trigger LANGUAGE plpgsql AS $$

BEGIN
  v_prev := NEW.old_status; -- 이전 상태(검증 완료)
  UPDATE compartments
     SET status=NEW.new_status
   WHERE compartment_id=NEW.compartment_id; -- 스냅샷 갱신

  VALUES (NEW.actor_user_id, 'RESOURCE_STATUS_CHANGED',
          'COMPARTMENT', NEW.compartment_id,
          'compartment status changed',
          jsonb_build_object('old', v_prev, 'new', NEW.new_status,
                             'reason', NEW.reason, 'event_id', NEW.event_id));
  RETURN NEW;
END; $$; -- 함수

-- 만료 세션 일괄 종료(배치 로그 포함)
CREATE FUNCTION close_expired_sessions()
RETURNS integer LANGUAGE plpgsql AS $$
DECLARE v_cnt integer;
BEGIN
  UPDATE inspection_sessions
     SET state='CLOSED', ended_at=now()
   WHERE state='ACTIVE' AND expires_at < now(); -- 만료 종료
  GET DIAGNOSTICS v_cnt = ROW_COUNT; -- 처리 수
  IF v_cnt > 0 THEN
    INSERT INTO event_log(action, entity_kind, message, meta)
    VALUES ('SESSION_EXPIRED_BATCH', 'INSPECTION_SESSION',
            'Expired sessions closed',
            jsonb_build_object('count', v_cnt));
  END IF;
  RETURN v_cnt;
END; $$; -- 함수

-- 방 배정 가드(용량·성별·만실) 점유 기반
CREATE FUNCTION tg_ura_guard()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_cap int; v_floor int; v_floor_gender text; v_user_gender text;
DECLARE v_active int;
BEGIN
  SELECT r.capacity, r.floor_no, r.occupied
    INTO v_cap, v_floor, v_active
  FROM rooms r
  WHERE r.room_id=NEW.room_id
  FOR UPDATE; -- 방 행 잠금

  SELECT f.gender INTO v_floor_gender
  FROM floors f WHERE f.floor_no=v_floor;
  SELECT u.gender INTO v_user_gender
  FROM users u WHERE u.user_id=NEW.user_id;

  IF NEW.personal_no > v_cap THEN
    RAISE EXCEPTION 'personal_no % > capacity %', NEW.personal_no, v_cap;
  END IF;
  IF v_floor_gender <> v_user_gender THEN
    RAISE EXCEPTION 'gender mismatch on floor %', v_floor;
  END IF;
  IF v_active >= v_cap THEN
    RAISE EXCEPTION 'room % is full', NEW.room_id;
  END IF;

  RETURN NEW;
END; $$; -- 함수
CREATE TRIGGER trg_ura_guard
BEFORE INSERT OR UPDATE OF room_id, personal_no, revoked_at
ON user_room_assignments
FOR EACH ROW WHEN (NEW.revoked_at IS NULL)
EXECUTE FUNCTION tg_ura_guard(); -- 활성 배정만

-- 점유 카운트 적용 트리거
CREATE FUNCTION tg_ura_occupancy_apply()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF TG_OP='INSERT' THEN
    IF NEW.revoked_at IS NULL THEN
      UPDATE rooms SET occupied=occupied+1 WHERE room_id=NEW.room_id;
    END IF;
  ELSIF TG_OP='UPDATE' THEN
    IF OLD.revoked_at IS NULL AND NEW.revoked_at IS NOT NULL THEN
      UPDATE rooms SET occupied=occupied-1 WHERE room_id=OLD.room_id;
    ELSIF OLD.revoked_at IS NOT NULL AND NEW.revoked_at IS NULL THEN
      UPDATE rooms SET occupied=occupied+1 WHERE room_id=NEW.room_id;
    ELSIF OLD.revoked_at IS NULL AND NEW.revoked_at IS NULL
          AND OLD.room_id<>NEW.room_id THEN
      UPDATE rooms SET occupied=occupied-1 WHERE room_id=OLD.room_id;
      UPDATE rooms SET occupied=occupied+1 WHERE room_id=NEW.room_id;
    END IF;
  ELSE
    IF OLD.revoked_at IS NULL THEN
      UPDATE rooms SET occupied=occupied-1 WHERE room_id=OLD.room_id;
    END IF;
  END IF;
  RETURN NULL; -- AFTER 트리거, 반환값 미사용
END; $$; -- 함수
CREATE TRIGGER trg_ura_occupancy_apply
AFTER INSERT OR UPDATE OR DELETE ON user_room_assignments
FOR EACH ROW EXECUTE FUNCTION tg_ura_occupancy_apply(); -- 점유 반영

-- 칸 접근권한 가드(LIST 스코프·동일 층)
CREATE FUNCTION tg_cra_guard()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_scope text; v_cf int; v_rf int;
BEGIN
  SELECT c.access_scope, fr.floor_no
    INTO v_scope, v_cf
  FROM compartments c
  JOIN fridges fr ON fr.fridge_id=c.fridge_id
  WHERE c.compartment_id=NEW.compartment_id; -- 칸 스코프/층
  SELECT r.floor_no INTO v_rf
  FROM rooms r WHERE r.room_id=NEW.room_id; -- 호실 층
  IF v_scope <> 'LIST' THEN
    RAISE EXCEPTION 'access_scope must be LIST';
  END IF;
  IF v_cf <> v_rf THEN
    RAISE EXCEPTION 'compartment and room must be on same floor';
  END IF;
  RETURN NEW;
END; $$; -- 함수

-- 물품 소유자 자격 가드(지연 검사, 뷰 미의존)
CREATE FUNCTION tg_item_group_owner_guard()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE v_floor_comp int; v_scope text; v_room_id bigint; v_floor_user int;
DECLARE v_allowed int;
BEGIN
  SELECT fr.floor_no, c.access_scope
    INTO v_floor_comp, v_scope
  FROM compartments c
  JOIN fridges fr ON fr.fridge_id=c.fridge_id
  WHERE c.compartment_id=NEW.compartment_id; -- 칸 층/스코프

  SELECT a.room_id, r.floor_no
    INTO v_room_id, v_floor_user
  FROM user_room_assignments a
  JOIN rooms r ON r.room_id=a.room_id
  WHERE a.user_id=NEW.owner_user_id AND a.revoked_at IS NULL
  ORDER BY a.assigned_at DESC LIMIT 1; -- 최신 활성

  IF v_room_id IS NULL THEN
    RAISE EXCEPTION 'owner has no active assignment';
  END IF;
  IF v_floor_user <> v_floor_comp THEN
    RAISE EXCEPTION 'owner floor mismatch';
  END IF;

  IF v_scope='LIST' THEN
    SELECT count(*) INTO v_allowed
    FROM compartment_room_access
    WHERE compartment_id=NEW.compartment_id AND room_id=v_room_id;
    IF v_allowed=0 THEN
      RAISE EXCEPTION 'room not allowed for this compartment';
    END IF;
  END IF;

  RETURN NEW;
END; $$; -- 함수
CREATE CONSTRAINT TRIGGER trg_item_group_owner_guard
AFTER INSERT OR UPDATE OF owner_user_id, compartment_id ON item_groups
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION tg_item_group_owner_guard(); -- 지연 검사
```
