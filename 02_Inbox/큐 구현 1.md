#include <stdio.h>
#include <stdlib.h>

#define MAX_QUEUE 10 // 10진수 기준 버킷 개수

typedef struct Node {
    int data;
    struct Node* next;
} Node;

typedef struct {
    Node *front, *rear;
} QueueType;

// 큐 초기화
void init(QueueType* q) {
    q->front = q->rear = NULL;
}

// 큐가 비어있는지 확인
int is_empty(QueueType* q) {
    return (q->front == NULL);
}

// Enqueue (삽입)
void enqueue(QueueType* q, int item) {
    Node* node = (Node*)malloc(sizeof(Node));
    node->data = item;
    node->next = NULL;
    if (is_empty(q)) {
        q->front = node;
        q->rear = node;
    } else {
        q->rear->next = node;
        q->rear = node;
    }
}

// Dequeue (삭제)
int dequeue(QueueType* q) {
    if (is_empty(q)) return -1;
    Node* temp = q->front;
    int data = temp->data;
    q->front = q->front->next;
    if (q->front == NULL) q->rear = NULL;
    free(temp);
    return data;
}

// 기수 정렬 (LSD)
void radix_sort(int list[], int n) {
    int i, bucket, digit, factor = 1;
    QueueType queues[MAX_QUEUE]; // 버킷 배열

    for (bucket = 0; bucket < MAX_QUEUE; bucket++) {
        init(&queues[bucket]);
    }

    // 10진수 가정, 최대 자릿수만큼 반복 (여기서는 예시로 4자리 가정)
    // 실제로는 max 값을 찾아 자릿수 계산 필요
    for (digit = 0; digit < 4; digit++) { 
        // 1. 분배 (Distribution)
        for (i = 0; i < n; i++) {
            // 현재 자릿수의 값에 해당하는 버킷에 넣음
            int bucket_idx = (list[i] / factor) % 10;
            enqueue(&queues[bucket_idx], list[i]);
        }

        // 2. 수집 (Collection)
        i = 0;
        for (bucket = 0; bucket < MAX_QUEUE; bucket++) {
            while (!is_empty(&queues[bucket])) {
                list[i++] = dequeue(&queues[bucket]);
            }
        }
        factor *= 10; // 다음 자릿수로 이동
    }
}
